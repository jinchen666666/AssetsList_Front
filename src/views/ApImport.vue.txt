<script setup>
    import { ref, computed, onMounted, onUnmounted, watch } from 'vue';
    import axios from 'axios';

    const API_BASE_URL = '/api';

    // state
    const companies = ref([]);                 // [{label, value}]
    const selectedCompanyId = ref('');
    const fileRef = ref(null);

    const uploading = ref(false);
    const loadingCompanies = ref(false);
    const loadingLogs = ref(false);

    const result = ref(null);
    const errorText = ref('');
    const showErpRaw = ref(false);

    const logs = ref([]);                      // all logs from server

    // filters (front-end)
    const filterCompanyId = ref('ALL');        // ALL or companyId
    const filterStatus = ref('ALL');           // ALL | SUCCESS | FAIL
    const keyword = ref('');                   // filename/message/docno

    // === 新增：分頁狀態與邏輯 ===
    const currentPage = ref(1);
    const pageSize = 10;

    // 當篩選條件改變時，自動回到第一頁
    watch([filterCompanyId, filterStatus, keyword], () => {
        currentPage.value = 1;
    });

    // === 新增：可搜尋下拉選單的狀態與邏輯 ===
    const isCompanyDropdownOpen = ref(false);
    const companySearchQuery = ref('');
    const companyDropdownRef = ref(null);

    const filteredDropdownCompanies = computed(() => {
        if (!companySearchQuery.value) return companies.value;
        const q = companySearchQuery.value.toLowerCase();
        return companies.value.filter(c =>
            (c.label || '').toLowerCase().includes(q) ||
            (c.value || '').toLowerCase().includes(q)
        );
    });

    const selectCompany = (c) => {
        selectedCompanyId.value = c.value;
        companySearchQuery.value = `${c.label} (${c.value})`;
        isCompanyDropdownOpen.value = false;
        errorText.value = '';
    };

    const toggleCompanyDropdown = () => {
        if (loadingCompanies.value || uploading.value) return;
        isCompanyDropdownOpen.value = !isCompanyDropdownOpen.value;
        if (isCompanyDropdownOpen.value) {
            companySearchQuery.value = ''; // 展開時清空，方便重新搜尋
        } else {
            // 收合時若未選擇，則恢復顯示目前選取的公司
            const found = companies.value.find(c => c.value === selectedCompanyId.value);
            companySearchQuery.value = found ? `${found.label} (${found.value})` : '';
        }
    };

    const closeDropdownClickOutside = (e) => {
        if (companyDropdownRef.value && !companyDropdownRef.value.contains(e.target)) {
            if (isCompanyDropdownOpen.value) {
                isCompanyDropdownOpen.value = false;
                const found = companies.value.find(c => c.value === selectedCompanyId.value);
                companySearchQuery.value = found ? `${found.label} (${found.value})` : '';
            }
        }
    };
    // ========================================

    // computed
    const selectedCompanyLabel = computed(() => {
        const found = companies.value.find(c => c.value === selectedCompanyId.value);
        return found ? found.label : '';
    });

    const canImport = computed(() => {
        return !!fileRef.value && !!selectedCompanyId.value && !uploading.value;
    });

    const stats = computed(() => {
        const all = logs.value?.length || 0;
        const ok = (logs.value || []).filter(x => x.success).length;
        return { all, ok, fail: all - ok };
    });

    const filteredLogs = computed(() => {
        let data = logs.value || [];

        if (filterCompanyId.value !== 'ALL') {
            data = data.filter(x => x.companyId === filterCompanyId.value);
        }

        if (filterStatus.value !== 'ALL') {
            const want = filterStatus.value === 'SUCCESS';
            data = data.filter(x => !!x.success === want);
        }

        const k = (keyword.value || '').trim().toLowerCase();
        if (k) {
            data = data.filter(x => {
                const a = (x.originalFileName || '').toLowerCase();
                const b = (x.message || '').toLowerCase();
                const c = (x.erpDocNo || '').toLowerCase();
                return a.includes(k) || b.includes(k) || c.includes(k);
            });
        }

        return data;
    });

    // === 新增：計算分頁資料 ===
    const totalPages = computed(() => Math.ceil((filteredLogs.value?.length || 0) / pageSize));

    const paginatedLogs = computed(() => {
        const start = (currentPage.value - 1) * pageSize;
        const end = start + pageSize;
        return filteredLogs.value?.slice(start, end) || [];
    });
    // ========================================

    // api
    const loadCompanies = async () => {
        loadingCompanies.value = true;
        try {
            const res = await axios.get(`${API_BASE_URL}/ap/companies`);
            companies.value = res.data || [];

            // default: prefer HDRE
            if (!selectedCompanyId.value && companies.value.length) {
                const hdre = companies.value.find(x => x.value === 'HDRE');
                const defaultComp = hdre || companies.value[0];
                selectedCompanyId.value = defaultComp.value;
                companySearchQuery.value = `${defaultComp.label} (${defaultComp.value})`; // 同步搜尋框顯示
            } else if (selectedCompanyId.value) {
                const found = companies.value.find(x => x.value === selectedCompanyId.value);
                if (found) companySearchQuery.value = `${found.label} (${found.value})`;
            }
        } catch (e) {
            errorText.value = '載入公司別失敗，請刷新或聯繫資訊人員';
        } finally {
            loadingCompanies.value = false;
        }
    };

    const loadMyLogs = async () => {
        loadingLogs.value = true;
        try {
            const res = await axios.get(`${API_BASE_URL}/ap/import/logs`);
            logs.value = res.data || [];
        } catch (e) {
            console.warn('load logs failed', e);
        } finally {
            loadingLogs.value = false;
        }
    };

    const downloadTemplate = () => {
        window.open(`${API_BASE_URL}/ap/import/template`, '_blank');
    };

    const onFileChange = (e) => {
        const f = e.target.files?.[0];
        fileRef.value = f || null;
        result.value = null;
        errorText.value = '';
        showErpRaw.value = false;
    };

    const resetForm = () => {
        fileRef.value = null;
        result.value = null;
        errorText.value = '';
        showErpRaw.value = false;
        const input = document.getElementById('ap-import-file');
        if (input) input.value = '';
    };

    const upload = async () => {
        if (!selectedCompanyId.value) {
            errorText.value = '請先選擇公司別';
            return;
        }
        if (!fileRef.value) {
            errorText.value = '請先選擇 Excel 檔案';
            return;
        }

        uploading.value = true;
        result.value = null;
        errorText.value = '';
        showErpRaw.value = false;

        try {
            const form = new FormData();
            form.append('file', fileRef.value);
            form.append('companyId', selectedCompanyId.value);

            const res = await axios.post(`${API_BASE_URL}/ap/import`, form, {
                headers: { 'Content-Type': 'multipart/form-data' }
            });

            result.value = res.data;
            currentPage.value = 1; // 匯入成功後跳回第一頁查看最新紀錄
            await loadMyLogs();
        } catch (err) {
            if (err?.response?.data) {
                result.value = err.response.data;
            } else {
                errorText.value = err?.message || '上傳失敗';
            }
            await loadMyLogs();
        } finally {
            uploading.value = false;
        }
    };


    const clearFilters = () => {
        filterCompanyId.value = 'ALL';
        filterStatus.value = 'ALL';
        keyword.value = '';
    };

    const formatTime = (timeStr) => {
        if (!timeStr) return '-';
        // 將 "2024-03-05T10:21:30" 簡化為 "2024-03-05 10:21"
        return timeStr.replace('T', ' ').substring(0, 16);
    };

    onMounted(async () => {
        document.addEventListener('click', closeDropdownClickOutside); // 註冊點擊外部事件
        await loadCompanies();
        await loadMyLogs();
    });

    onUnmounted(() => {
        document.removeEventListener('click', closeDropdownClickOutside); // 移除事件避免 memory leak
    });
</script>

                                        <path class="opacity-75" fill="currentColor" d="M4 12a8 8 0 018-8V0C5.373 0 0 5.373 0 12h4zm2 5.291A7.962 7.962 0 014 12H0c0 3.042 1.135 5.824 3 7.938l3-2.647z"></path>
                                    </svg>
                                    <svg v-else xmlns="http://www.w3.org/2000/svg" class="w-4 h-4 text-slate-400 transition-transform duration-200" :class="{'rotate-180': isCompanyDropdownOpen}" viewBox="0 0 20 20" fill="currentColor">
                                        <path fill-rule="evenodd" d="M5.293 7.293a1 1 0 011.414 0L10 10.586l3.293-3.293a1 1 0 111.414 1.414l-4 4a1 1 0 01-1.414 0l-4-4a1 1 0 010-1.414z" clip-rule="evenodd" />
