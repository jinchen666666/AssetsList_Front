<script setup>
    import { ref, watch, computed } from 'vue';
    import axios from 'axios';

    import SystemConfigGeneral from '../components/admin/SystemConfigGeneral.vue';
    import SystemConfigApImport from '../components/admin/SystemConfigApImport.vue';
    import SystemConfigAssetCode from '../components/admin/SystemConfigAssetCode.vue';

    const props = defineProps({
        currentUser: Object,
        permissions: {
            type: Array,
            default: () => []
        }
    });

    const PERM = {
        AdminSystemConfig: 'Admin.SystemConfig'
    };

    const permSet = computed(() => new Set((props.permissions || []).map(String)));
    const hasPerm = (key) => !key || permSet.value.has(key);
    const canAdmin = computed(() => hasPerm(PERM.AdminSystemConfig));

    const API_BASE_URL = '/api';
    const systemConfigs = ref([]);
    const adminLoading = ref(false);
    const activeTab = ref('general');

    const toastStatus = ref({ show: false, message: '', success: true });
    const showToast = (message, success = true) => {
        toastStatus.value = { show: true, message, success };
        setTimeout(() => { toastStatus.value.show = false; }, 3000);
    };

    const getKey = (c) => c.configKey || c.ConfigKey || '';
    const getValue = (c) => (c.configValue ?? c.ConfigValue ?? '');
    const getDesc = (c) => c.description || c.Description || '';
    const setValue = (c, val) => { c.configValue = val; };
    const setDesc = (c, val) => { c.description = val; };

    const ensureConfigExists = (key, desc, defaultValue = '') => {
        const found = systemConfigs.value.find(x => getKey(x) === key);
        if (!found) {
            systemConfigs.value.push({
                configKey: key,
                configValue: defaultValue,
                description: desc
            });
        } else {
            if (!getDesc(found) && desc) setDesc(found, desc);
            if ((found.configValue ?? found.ConfigValue) == null) setValue(found, defaultValue);
        }
    };

    const fetchAdminConfigs = async () => {
        if (!canAdmin.value) return;

        adminLoading.value = true;
        try {
            const res = await axios.get(`${API_BASE_URL}/admin/config`);
            systemConfigs.value = res.data || [];

            ensureConfigExists('Transfer_BlockPeriod', '移轉單關帳限制 (每月起始日-次月結束日)', '25-1');
            ensureConfigExists('AssetNo_Prefixes', '資產編號 Prefix 限制', '');

            ensureConfigExists(
                'ApImport_CompanyAllowMap',
                '應付匯入公司別授權 (工號=公司1,公司2；每行一筆)',
                ''
            );

            ensureConfigExists('AssetCode.GeneralAffairsOptions', '格式 代碼|中文名稱，例如 99|總務', '99|總務');
            ensureConfigExists('AssetCode.AssetCategoryOptions', '格式 代碼|中文名稱，例如 04|家具', '04|家具');
            ensureConfigExists('AssetCode.ItemCategoryOptions', '格式 代碼|中文名稱，例如 TB|桌子', 'TB|桌子');
            ensureConfigExists('AssetCode.Default.GeneralAffairs', '資產編號預設總務識別碼', '99');
            ensureConfigExists('AssetCode.Default.AssetCategory', '資產編號預設資產類別碼', '04');
            ensureConfigExists('AssetCode.Default.ItemCategory', '資產編號預設品項類別碼', 'TB');
        } catch (e) {
            console.error('Admin Config fetch failed', e);
            showToast('載入參數失敗，請稍後再試', false);
        } finally {
            adminLoading.value = false;
        }
    };

    const saveConfigPayload = async (payload) => {
        await axios.post(`${API_BASE_URL}/admin/config`, payload);
    };

    const upsertLocalConfig = (key, value, desc) => {
        const found = systemConfigs.value.find(x => getKey(x) === key);
        if (found) {
            setValue(found, value);
            setDesc(found, desc);
        } else {
            systemConfigs.value.push({
                configKey: key,
                configValue: value,
                description: desc
            });
        }
    };

    watch(() => props.currentUser, (newVal) => {
        if (newVal && canAdmin.value) {
            fetchAdminConfigs();
        }
    }, { immediate: true });

    watch(canAdmin, (ok) => {
        if (ok && props.currentUser) fetchAdminConfigs();
    });
</script>

<template>
    <div class="flex flex-col gap-5 pb-4 w-full min-w-0 relative">

        <div class="fixed top-6 right-6 z-[9999] transition-all duration-300 transform"
             :class="toastStatus.show ? 'translate-y-0 opacity-100' : '-translate-y-4 opacity-0 pointer-events-none'">
            <div class="px-4 py-3 rounded-lg shadow-lg border flex items-center gap-3"
                 :class="toastStatus.success ? 'bg-emerald-50 border-emerald-200 text-emerald-800' : 'bg-red-50 border-red-200 text-red-800'">
                <svg v-if="toastStatus.success" xmlns="http://www.w3.org/2000/svg" class="w-5 h-5 text-emerald-500" viewBox="0 0 20 20" fill="currentColor">
                    <path fill-rule="evenodd" d="M10 18a8 8 0 100-16 8 8 0 000 16zm3.707-9.293a1 1 0 00-1.414-1.414L9 10.586 7.707 9.293a1 1 0 00-1.414 1.414l2 2a1 1 0 001.414 0l4-4z" clip-rule="evenodd" />
                </svg>
                <svg v-else xmlns="http://www.w3.org/2000/svg" class="w-5 h-5 text-red-500" viewBox="0 0 20 20" fill="currentColor">
                    <path fill-rule="evenodd" d="M10 18a8 8 0 100-16 8 8 0 000 16zM8.707 7.293a1 1 0 00-1.414 1.414L8.586 10l-1.293 1.293a1 1 0 101.414 1.414L10 11.414l1.293 1.293a1 1 0 001.414-1.414L11.414 10l1.293-1.293a1 1 0 00-1.414-1.414L10 8.586 8.707 7.293z" clip-rule="evenodd" />
                </svg>
                <span class="font-bold text-sm">{{ toastStatus.message }}</span>
            </div>
        </div>

        <div class="p-4 bg-purple-50/60 border border-purple-100 rounded-xl shadow-sm flex items-start gap-3 text-slate-600 w-full">
            <svg xmlns="http://www.w3.org/2000/svg" class="w-5 h-5 text-purple-500 mt-0.5 shrink-0" fill="none" viewBox="0 0 24 24" stroke="currentColor" stroke-width="2">
                <path stroke-linecap="round" stroke-linejoin="round" d="M10.325 4.317c.426-1.756 2.924-1.756 3.35 0a1.724 1.724 0 002.573 1.066c1.543-.94 3.31.826 2.37 2.37a1.724 1.724 0 001.065 2.572c1.756.426 1.756 2.924 0 3.35a1.724 1.724 0 00-1.066 2.573c.94 1.543-.826 3.31-2.37 2.37a1.724 1.724 0 00-2.572 1.065c-.426 1.756-2.924 1.756-3.35 0a1.724 1.724 0 00-2.573-1.066c-1.543.94-3.31-.826-2.37-2.37a1.724 1.724 0 00-1.065-2.572c-1.756-.426-1.756-2.924 0-3.35a1.724 1 0 001.066-2.573c-.94-1.543.826-3.31 2.37-2.37.996.608 2.296.07 2.572-1.065z" />
                <path stroke-linecap="round" stroke-linejoin="round" d="M15 12a3 3 0 11-6 0 3 3 0 016 0z" />
            </svg>
            <div>
                <h3 class="font-bold mb-1 text-purple-700">參數設定中心</h3>
                <ul class="list-disc list-inside text-sm space-y-1 text-slate-600">
                    <li>將系統參數拆成子頁面，便於維護與後續權限控管。</li>
                    <li>請先切換子選單，再針對對應區塊進行編輯與儲存。</li>
                </ul>
            </div>
        </div>

        <div v-if="!canAdmin" class="p-8 text-center text-slate-500 bg-white/40 rounded-xl border border-white/60 shadow-sm w-full">
            <svg xmlns="http://www.w3.org/2000/svg" fill="none" viewBox="0 0 24 24" stroke-width="1.5" stroke="currentColor" class="w-12 h-12 mx-auto text-slate-300 mb-3">
                <path stroke-linecap="round" stroke-linejoin="round" d="M16.5 10.5V6.75a4.5 4.5 0 10-9 0v3.75m-.75 11.25h10.5a2.25 2.25 0 002.25-2.25v-6.75a2.25 2.25 0 00-2.25-2.25H6.75a2.25 2.25 0 00-2.25 2.25v6.75a2.25 2.25 0 002.25 2.25z" />
            </svg>
            你沒有「系統管理」權限
        </div>

        <div v-else class="rounded-xl border border-white/60 bg-white/40 shadow-sm p-5 w-full min-w-0">
            <div class="flex flex-col md:flex-row md:items-center md:justify-between gap-4 mb-5 pb-4 border-b border-slate-200/50">
                <div>
                    <h2 class="text-lg font-bold text-slate-800 flex items-center gap-2">
                        <svg xmlns="http://www.w3.org/2000/svg" class="w-5 h-5 text-purple-600" fill="none" viewBox="0 0 24 24" stroke="currentColor" stroke-width="2">
                            <path stroke-linecap="round" stroke-linejoin="round" d="M8 9l3 3-3 3m5 0h3M5 20h14a2 2 0 002-2V6a2 2 0 00-2-2H5a2 2 0 00-2 2v12a2 2 0 002 2z" />
                        </svg>
                        系統參數設定
                    </h2>
                </div>

                <div class="flex items-center gap-2">
                    <button @click="fetchAdminConfigs" :disabled="adminLoading"
                            class="px-3 py-2 bg-white/60 hover:bg-white text-slate-700 rounded-lg shadow-sm border border-slate-200 transition-all flex items-center gap-2 hover:shadow-md text-sm font-medium disabled:opacity-50">
                        <svg xmlns="http://www.w3.org/2000/svg" class="w-4 h-4 text-slate-500" :class="{ 'animate-spin text-purple-600': adminLoading }" fill="none" viewBox="0 0 24 24" stroke="currentColor" stroke-width="2">
                            <path stroke-linecap="round" stroke-linejoin="round" d="M4 4v5h.582m15.356 2A8.001 8.001 0 004.582 9m0 0H9m11 11v-5h-.581m0 0a8.003 8.003 0 01-15.357-2m15.357 2H15" />
                        </svg>
                        重新整理
                    </button>
                </div>
            </div>

            <div v-if="adminLoading" class="p-8 text-center text-slate-500 flex justify-center items-center gap-2">
                <svg class="animate-spin h-5 w-5 text-purple-500" xmlns="http://www.w3.org/2000/svg" fill="none" viewBox="0 0 24 24">
                    <circle class="opacity-25" cx="12" cy="12" r="10" stroke="currentColor" stroke-width="4"></circle>
                    <path class="opacity-75" fill="currentColor" d="M4 12a8 8 0 018-8V0C5.373 0 0 5.373 0 12h4zm2 5.291A7.962 7.962 0 014 12H0c0 3.042 1.135 5.824 3 7.938l3-2.647z"></path>
                </svg>
                資料載入中...
            </div>

            <template v-else>
                <div class="flex flex-wrap gap-2 mb-5">
                    <button @click="activeTab = 'general'"
                            class="px-4 py-2 rounded-lg text-sm font-medium border transition-all"
                            :class="activeTab === 'general'
                            ? 'bg-purple-600 text-white border-purple-600 shadow-sm'
                            : 'bg-white/70 text-slate-700 border-slate-200 hover:bg-white'">
                        一般參數
                    </button>

                    <button @click="activeTab = 'ap-import'"
                            class="px-4 py-2 rounded-lg text-sm font-medium border transition-all"
                            :class="activeTab === 'ap-import'
                            ? 'bg-purple-600 text-white border-purple-600 shadow-sm'
                            : 'bg-white/70 text-slate-700 border-slate-200 hover:bg-white'">
                        AP 授權設定
                    </button>

                    <button @click="activeTab = 'asset-code'"
                            class="px-4 py-2 rounded-lg text-sm font-medium border transition-all"
                            :class="activeTab === 'asset-code'
                            ? 'bg-purple-600 text-white border-purple-600 shadow-sm'
                            : 'bg-white/70 text-slate-700 border-slate-200 hover:bg-white'">
                        資產編碼設定
                    </button>
                </div>

                <SystemConfigGeneral v-if="activeTab === 'general'"
                                     :system-configs="systemConfigs"
                                     :can-admin="canAdmin"
                                     @save-config="async (payload, done) => { try { await saveConfigPayload(payload); done(true); showToast('設定已成功儲存！', true); } catch (e) { console.error(e); done(false); showToast('儲存失敗，請檢查網路或權限', false); } }" />

                <SystemConfigApImport v-else-if="activeTab === 'ap-import'"
                                      :system-configs="systemConfigs"
                                      :can-admin="canAdmin"
                                      @save-config="async (payload, done) => { try { await saveConfigPayload(payload); upsertLocalConfig(payload.ConfigKey, payload.ConfigValue, payload.Description); done(true); showToast('設定已成功儲存！', true); } catch (e) { console.error(e); done(false); showToast('儲存失敗，請檢查網路或權限', false); } }" />

                <SystemConfigAssetCode v-else
                                       :system-configs="systemConfigs"
                                       :can-admin="canAdmin"
                                       @save-batch="async (payloads, done) => {
                        try {
                            for (const payload of payloads) {
                                await saveConfigPayload(payload);
                                upsertLocalConfig(payload.ConfigKey, payload.ConfigValue, payload.Description);
                            }
                            done(true);
                            showToast('資產編碼設定已成功儲存！', true);
                        } catch (e) {
                            console.error(e);
                            done(false);
                            showToast('資產編碼設定儲存失敗，請檢查網路或權限', false);
                        }
                    }" />
            </template>
        </div>
    </div>
</template>
