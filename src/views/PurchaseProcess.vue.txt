<script setup>
    import { ref, computed, onMounted, watch } from 'vue'
    import axios from 'axios'
    import PurchaseAssetRequestModal from '../components/PurchaseAssetRequestModal.vue'
    import PurchaseSplitAssetRequestModal from '../components/PurchaseSplitAssetRequestModal.vue'
    import DateRangePicker from '../components/common/DateRangePicker.vue'

    const props = defineProps({
        currentUser: { type: Object, default: null },
        myPermissions: { type: Array, default: () => [] },
        draftId: { type: [Number, String], default: null }
    })

    const emit = defineEmits(['change-view'])

    const purchases = ref([])
    const isLoading = ref(false)
    const isProcessing = ref(false)
    const showSingleModal = ref(false)
    const systemConfigs = ref([])
    const showSplitModal = ref(false)
    const isFallbackData = ref(false)
    const pendingCount = ref(0)
    const isRefreshingPending = ref(false)
    const currentDraftId = ref(null)
    const currentSplitDraftId = ref(null)

    const PURCHASE_PROCESS_CACHE_KEY = 'ams.purchaseProcess.cache.v1'
    const hasFetchedPending = ref(false)
    const lastFetchedAt = ref('')

    const queryStartDate = ref('')
    const queryEndDate = ref('')

    const columnFilters = ref({
        docNo: '',
        itemNo: '',
        itemName: '',
        itemSpec: '',
        vendorName: '',
        incomingDate: '',
        //qty: '',
        //unit: '',
        keeperName: '',
        keeperDept: '',
        location: ''
    })

    const toYyyyMMdd = (value) => {
        if (!value) return ''
        return value.replaceAll('-', '')
    }

    const setDefaultQueryRange = () => {
        const today = new Date()
        const start = new Date()
        start.setMonth(start.getMonth() - 6)

        const formatDateInput = (date) => {
            const yyyy = date.getFullYear()
            const mm = String(date.getMonth() + 1).padStart(2, '0')
            const dd = String(date.getDate()).padStart(2, '0')
            return `${yyyy}-${mm}-${dd}`
        }

        if (!queryStartDate.value) queryStartDate.value = formatDateInput(start)
        if (!queryEndDate.value) queryEndDate.value = formatDateInput(today)
    }

    const getDateRangeError = () => {
        if (!queryStartDate.value || !queryEndDate.value) {
            return '請選擇起日與訖日'
        }

        const start = new Date(queryStartDate.value)
        const end = new Date(queryEndDate.value)

        if (Number.isNaN(start.getTime()) || Number.isNaN(end.getTime())) {
            return '日期格式錯誤'
        }

        if (start > end) {
            return '起日不可大於訖日'
        }

        const maxEnd = new Date(start)
        maxEnd.setMonth(maxEnd.getMonth() + 6)

        if (end > maxEnd) {
            return '查詢日期區間不可超過 6 個月'
        }

        return ''
    }

    const getSourceKey = (item) => {
        return [
            item?.docTypeNo || '',
            item?.docNo || '',
            item?.seq || ''
        ].map(x => String(x).trim().toUpperCase()).join('|')
    }

    const getDisplayLocation = (item) => {
        return (
            item?.displayLocation ||
            item?.storageLocation ||
            item?.locationNo ||
            item?.warehouseName ||
            item?.warehouseNo ||
            item?.location ||
            ''
        )
    }

    const normalizeFilterText = (value) => {
        return String(value ?? '')
            .trim()
            .toLowerCase()
            .replaceAll('/', '')
            .replaceAll('-', '')
            .replaceAll(' ', '')
    }

    const includesFilter = (value, filter) => {
        const f = normalizeFilterText(filter)
        if (!f) return true
        return normalizeFilterText(value).includes(f)
    }

    const clearColumnFilters = () => {
        Object.keys(columnFilters.value).forEach(key => {
            columnFilters.value[key] = ''
        })
    }

    const hasColumnFilter = computed(() =>
        Object.values(columnFilters.value).some(x => String(x || '').trim())
    )

    const formatDate = (dateStr) => {
        if (!dateStr || String(dateStr).length !== 8) return dateStr || ''
        const value = String(dateStr)
        return `${value.substring(0, 4)}/${value.substring(4, 6)}/${value.substring(6, 8)}`
    }

    const filteredPurchases = computed(() => {
        return purchases.value.filter(item =>
            includesFilter(item.docNo, columnFilters.value.docNo) &&
            includesFilter(item.itemNo, columnFilters.value.itemNo) &&
            includesFilter(item.itemName, columnFilters.value.itemName) &&
            includesFilter(item.itemSpec, columnFilters.value.itemSpec) &&
            includesFilter(`${item.vendorName || ''} ${item.vendorId || ''}`, columnFilters.value.vendorName) &&
            includesFilter(item.vendorName, columnFilters.value.vendorName) &&
            includesFilter(formatDate(item.incomingDate), columnFilters.value.incomingDate) &&
            //includesFilter(item.qty, columnFilters.value.qty) &&
            //includesFilter(item.unit, columnFilters.value.unit) &&
            includesFilter(item.keeperName, columnFilters.value.keeperName) &&
            includesFilter(item.keeperDept, columnFilters.value.keeperDept) &&
            includesFilter(getDisplayLocation(item), columnFilters.value.location)
        )
    })

    const savePurchaseProcessCache = () => {
        try {
            const cache = {
                queryStartDate: queryStartDate.value,
                queryEndDate: queryEndDate.value,
                purchases: purchases.value || [],
                pendingCount: pendingCount.value || 0,
                isFallbackData: !!isFallbackData.value,
                hasFetchedPending: !!hasFetchedPending.value,
                lastFetchedAt: lastFetchedAt.value || ''
            }

            sessionStorage.setItem(PURCHASE_PROCESS_CACHE_KEY, JSON.stringify(cache))
        } catch (e) {
            console.warn('儲存 ERP 進貨單查詢快取失敗:', e)
        }
    }

    const restorePurchaseProcessCache = () => {
        try {
            const raw = sessionStorage.getItem(PURCHASE_PROCESS_CACHE_KEY)
            if (!raw) return false

            const cache = JSON.parse(raw)
            if (!cache) return false

            queryStartDate.value = cache.queryStartDate || queryStartDate.value
            queryEndDate.value = cache.queryEndDate || queryEndDate.value

            purchases.value = Array.isArray(cache.purchases)
                ? cache.purchases.map(x => ({ ...x, selected: false }))
                : []

            pendingCount.value = Number(cache.pendingCount || purchases.value.length || 0)
            isFallbackData.value = !!cache.isFallbackData
            hasFetchedPending.value = !!cache.hasFetchedPending
            lastFetchedAt.value = cache.lastFetchedAt || ''

            return hasFetchedPending.value
        } catch (e) {
            console.warn('還原 ERP 進貨單查詢快取失敗:', e)
            return false
        }
    }

    const clearPurchaseProcessCache = () => {
        try {
            sessionStorage.removeItem(PURCHASE_PROCESS_CACHE_KEY)
        } catch (e) {
            console.warn('清除 ERP 進貨單查詢快取失敗:', e)
        }
    }

    const removeProcessedItemsFromCurrentList = (items) => {
        const keys = new Set((items || []).map(getSourceKey).filter(Boolean))
        if (keys.size === 0) return

        purchases.value = purchases.value.filter(x => !keys.has(getSourceKey(x)))
        pendingCount.value = purchases.value.length
        hasFetchedPending.value = true

        savePurchaseProcessCache()
    }

    const fetchPendingPurchases = async () => {
        const rangeError = getDateRangeError()
        if (rangeError) {
            alert(rangeError)
            return
        }

        isLoading.value = true

        try {
            const response = await axios.get('/api/Purchase/pending', {
                params: {
                    startDate: toYyyyMMdd(queryStartDate.value),
                    endDate: toYyyyMMdd(queryEndDate.value)
                },
                withCredentials: true
            })

            const result = response.data || {}
            const items = Array.isArray(result.items) ? result.items : []

            isFallbackData.value = !!result.isFallback
            pendingCount.value = Number(result.count || items.length || 0)
            hasFetchedPending.value = true
            lastFetchedAt.value = new Date().toISOString()

            purchases.value = items.map(item => ({
                ...item,
                selected: false
            }))

            clearColumnFilters()
            savePurchaseProcessCache()
        } catch (error) {
            console.error('載入待處理進貨單失敗:', error)

            isFallbackData.value = false
            pendingCount.value = 0
            purchases.value = []
            hasFetchedPending.value = false
            lastFetchedAt.value = ''

            clearColumnFilters()
            clearPurchaseProcessCache()

            alert(error?.response?.data?.message || error?.response?.data || error?.message || '載入待處理進貨單失敗')
        } finally {
            isLoading.value = false
        }
    }

    const refreshPendingPurchases = async () => {
        isRefreshingPending.value = true
        try {
            await fetchPendingPurchases()
        } finally {
            isRefreshingPending.value = false
        }
    }

    const clearPendingQuery = async () => {
        queryStartDate.value = ''
        queryEndDate.value = ''

        setDefaultQueryRange()

        purchases.value = []
        pendingCount.value = 0
        isFallbackData.value = false
        hasFetchedPending.value = false
        lastFetchedAt.value = ''

        clearColumnFilters()
        clearPurchaseProcessCache()
    }

    const fetchConfigs = async () => {
        try {
            const res = await axios.get('/api/admin/config')
            systemConfigs.value = Array.isArray(res.data) ? res.data : []
        } catch (e) {
            console.error(e)
            systemConfigs.value = []
        }
    }

    const selectedItems = computed(() => purchases.value.filter(x => x.selected))

    const isAllSelected = computed(() =>
        filteredPurchases.value.length > 0 &&
        filteredPurchases.value.every(x => x.selected)
    )

    const toggleSelectAll = (checked) => {
        filteredPurchases.value.forEach(item => {
            item.selected = checked
        })
    }

    const handleExpense = async () => {
        if (selectedItems.value.length === 0) {
            alert('請先勾選資料')
            return
        }

        if (!confirm(`確定將 ${selectedItems.value.length} 筆資料排除不需處理？排除後將不再出現在待處理清單。`)) return

        isProcessing.value = true

        try {
            const processedItems = [...selectedItems.value]

            const payload = {
                items: processedItems.map(x => ({
                    docTypeNo: x.docTypeNo,
                    docNo: x.docNo,
                    seq: x.seq
                }))
            }

            await axios.post('/api/Purchase/expense', payload)

            alert('已列為不處理，查詢ERP進貨單不會再顯示')

            removeProcessedItemsFromCurrentList(processedItems)
        } catch (e) {
            console.error(e)
            alert(e?.response?.data?.message || e?.response?.data || e?.message || '排除不需處理失敗')
        } finally {
            isProcessing.value = false
        }
    }

    const openSingleAssetModal = () => {
        if (selectedItems.value.length === 0) {
            alert('請先勾選資料')
            return
        }

        currentDraftId.value = null
        currentSplitDraftId.value = null
        showSplitModal.value = false
        showSingleModal.value = true
    }

    const openSplitPlaceholder = () => {
        if (selectedItems.value.length === 0) {
            alert('請先勾選資料')
            return
        }

        currentSplitDraftId.value = null
        currentDraftId.value = null
        showSingleModal.value = false
        showSplitModal.value = true
    }

    const openDraftList = () => {
        emit('change-view', 'purchase-request-draft-list')
    }

    const handleCloseSingleModal = () => {
        showSingleModal.value = false
        currentDraftId.value = null
    }

    const handleSingleModalSaved = async () => {
        const processedItems = [...selectedItems.value]

        handleCloseSingleModal()
        removeProcessedItemsFromCurrentList(processedItems)
    }

    const handleCloseSplitModal = () => {
        showSplitModal.value = false
        currentSplitDraftId.value = null
    }

    const handleSplitModalSaved = async () => {
        const processedItems = [...selectedItems.value]

        handleCloseSplitModal()
        removeProcessedItemsFromCurrentList(processedItems)
    }

    onMounted(async () => {
        setDefaultQueryRange()

        const restored = restorePurchaseProcessCache()
        if (!restored) {
            purchases.value = []
            pendingCount.value = 0
            isFallbackData.value = false
            hasFetchedPending.value = false
            lastFetchedAt.value = ''
        }

        await fetchConfigs()
    })

    watch(
        () => props.draftId,
        async (val) => {
            if (!val) return

            try {
                const res = await axios.get(`/api/purchase-request-draft/${val}`)
                const draft = res.data
                if (!draft) return

                if (draft.requestType === 'SplitAsset') {
                    currentSplitDraftId.value = val
                    showSplitModal.value = true
                    currentDraftId.value = null
                    showSingleModal.value = false
                    return
                }

                currentDraftId.value = val
                showSingleModal.value = true
                currentSplitDraftId.value = null
                showSplitModal.value = false
            } catch (e) {
                console.error(e)
                alert('載入待處理失敗')
            }
        },
        { immediate: true }
    )
</script>

    <div class="h-full flex flex-col">
        <div class="mb-4 p-4 bg-indigo-50/60 border border-indigo-100 rounded-xl shadow-sm flex items-start gap-3 text-slate-600 shrink-0">
            <svg xmlns="http://www.w3.org/2000/svg"
                 viewBox="0 0 20 20"
                 fill="currentColor"
                 class="w-5 h-5 text-indigo-500 mt-0.5 shrink-0">
                <path fill-rule="evenodd"
                      d="M18 10a8 8 0 11-16 0 8 8 0 0116 0zm-7-4a1 1 0 11-2 0 1 1 0 012 0zM9 9a1 1 0 000 2v3a1 1 0 001 1h1a1 1 0 100-2v-3a1 1 0 00-1-1H9z"
                      clip-rule="evenodd" />
            </svg>
            <div>
                <h3 class="font-bold mb-1 text-indigo-700">作業說明</h3>

                    <li>儲存待處理後，本頁不會自動重新查詢 ERP；只有按「查詢 ERP 進貨單」才會重新撈取資料。</li>
                </ul>

        </div>

        <div class="mb-4 p-4 rounded-xl border border-slate-200 bg-white/70 shadow-sm">
            <div class="grid grid-cols-1 md:grid-cols-2 xl:grid-cols-5 gap-4 items-end">
                <DateRangePicker v-model:start-date="queryStartDate"
                                 v-model:end-date="queryEndDate"
                                 :max-months="6"
                                 class="md:col-span-2 xl:col-span-3" />

                <div class="flex gap-2 xl:col-span-2">
                    <button type="button"
                            @click="fetchPendingPurchases"
                            :disabled="isLoading || isRefreshingPending || isProcessing"
                            class="px-4 py-2 rounded-lg bg-indigo-600 text-white text-sm font-medium hover:bg-indigo-700 disabled:opacity-50">
                        {{ isLoading ? '查詢中...' : '查詢 ERP 進貨單' }}
                    </button>

                    <button type="button"
                            @click="clearPendingQuery"
                            class="px-4 py-2 rounded-lg bg-white border border-slate-300 text-slate-700 text-sm font-medium hover:bg-slate-50">
                        清除
                    </button>
                </div>
            </div>
        </div>

        <div class="mb-6 flex justify-between items-center shrink-0">
            <h2 class="text-xl font-bold text-slate-800">ERP 進貨單轉列管資產處理</h2>

            <div class="flex gap-3 flex-wrap">
                <button @click="handleExpense"
                        :disabled="isProcessing || selectedItems.length === 0"
                        class="px-4 py-2 bg-rose-50 hover:bg-rose-100 text-rose-600 border border-rose-200 rounded-lg shadow-sm transition-all transform active:scale-95 disabled:opacity-50 font-medium">
                    排除不需處理
                </button>

                <button @click="openSingleAssetModal"
                        :disabled="isProcessing || selectedItems.length === 0"
                        class="px-4 py-2 bg-blue-600 hover:bg-blue-700 text-white rounded-lg shadow-md shadow-blue-300/50 transition-all transform active:scale-95 disabled:opacity-50 font-medium">
                    建立一筆資產
                </button>

                <button @click="openSplitPlaceholder"
                        :disabled="isProcessing || selectedItems.length === 0"
                        class="px-4 py-2 bg-indigo-500 hover:bg-indigo-600 text-white rounded-lg shadow-md shadow-indigo-300/50 transition-all transform active:scale-95 disabled:opacity-50 font-medium">
                    拆為多筆資產
                </button>

                <button @click="openDraftList"
                        class="px-4 py-2 bg-white/80 hover:bg-white text-slate-700 rounded-lg shadow-sm border border-slate-200 transition-all flex items-center gap-2 hover:shadow-md transform active:scale-95 font-medium">
                    待處理查詢
                </button>
            </div>
        </div>

        <div class="rounded-xl border border-white/60 bg-white/40 shadow-inner flex-1 flex flex-col overflow-hidden">
            <div class="p-4 border-b border-slate-200/50 flex items-center justify-between gap-3 flex-wrap bg-white/30">
                <div class="text-sm text-slate-500">
                    待處理筆數：
                    <span class="font-bold text-slate-700">{{ pendingCount }}</span>
                    <span v-if="hasColumnFilter" class="ml-2 text-slate-400">
                        / 篩選後 {{ filteredPurchases.length }} 筆
                    </span>
                </div>

                <div class="flex items-center gap-2">
                    <div v-if="hasFetchedPending"
                         class="px-3 py-1.5 rounded-lg text-sm font-medium border"
                         :class="isFallbackData
                            ? 'bg-amber-50 text-amber-700 border-amber-200'
                            : 'bg-emerald-50 text-emerald-700 border-emerald-200'">
                        {{ isFallbackData ? '目前顯示 Demo 資料' : '目前顯示 ERP 真實資料' }}
                    </div>

                    <div v-else
                         class="px-3 py-1.5 rounded-lg text-sm font-medium border bg-slate-50 text-slate-600 border-slate-200">
                        尚未查詢
                    </div>
                </div>
            </div>

            <div v-if="isLoading"
                 class="p-10 flex flex-col justify-center items-center text-slate-500 h-full">
                <div class="animate-spin border-4 border-indigo-500 border-t-transparent rounded-full w-10 h-10 mb-4"></div>
                <p class="font-medium tracking-wide">載入 ERP 待處理進貨單中...</p>
            </div>

            <div v-else-if="!hasFetchedPending"
                 class="p-10 flex flex-col justify-center items-center text-slate-400 h-full font-medium">
                <div class="text-slate-700 font-bold mb-1">尚未查詢 ERP 進貨單</div>
                <div class="text-sm text-slate-500">請先輸入起訖日期，然後按「查詢 ERP 進貨單」。</div>
            </div>

            <div v-else-if="purchases.length === 0"
                 class="p-10 flex justify-center items-center text-slate-400 h-full font-medium">
                目前沒有待處理的進貨單。
            </div>

            <div v-else class="flex-1 overflow-auto no-scrollbar">
                <table class="w-full min-w-[1520px] table-fixed divide-y divide-slate-200 relative">
                    <colgroup>
                        <col style="width:52px" />

                        <col style="width:105px" />

                        <col style="width:170px" />
                        <col style="width:150px" />
                        <col style="width:110px" />
                        <col style="width:80px" />
                        <col style="width:70px" />
                        <col style="width:120px" />
                        <col style="width:140px" />
                        <col style="width:173px" />
                    </colgroup>

                    <thead class="bg-white/95 backdrop-blur-md sticky top-0 z-10 border-b border-slate-200/50">
                        <tr>
                            <th class="px-2 py-3 text-center">
                                <input type="checkbox"
                                       @change="toggleSelectAll($event.target.checked)"
                                       :checked="isAllSelected"
                                       class="h-4 w-4">
                            </th>
                            <th class="table-th">進貨單號</th>
                            <th class="table-th">品號</th>
                            <th class="table-th">品名</th>
                            <th class="table-th">規格</th>
                            <th class="table-th">廠商</th>
                            <th class="table-th">驗收日期</th>
                            <th class="table-th">數量</th>
                            <th class="table-th">單位</th>
                            <th class="table-th">保管人</th>
                            <th class="table-th">保管部門</th>
                            <th class="table-th">放置地點</th>
                        </tr>

                        <tr class="bg-slate-50/80 border-t border-slate-100">
                            <th class="px-2 py-2"></th>

                            <th class="px-2 py-2">
                                <input v-model="columnFilters.docNo" placeholder="" class="filter-input" />
                            </th>
                            <th class="px-2 py-2">
                                <input v-model="columnFilters.itemNo" placeholder="" class="filter-input" />
                            </th>
                            <th class="px-2 py-2">
                                <input v-model="columnFilters.itemName" placeholder="" class="filter-input" />
                            </th>
                            <th class="px-2 py-2">
                                <input v-model="columnFilters.itemSpec" placeholder="" class="filter-input" />
                            </th>
                            <th class="px-2 py-2">
                                <input v-model="columnFilters.vendorName" placeholder="" class="filter-input" />
                            </th>
                            <th class="px-2 py-2">
                                <input v-model="columnFilters.incomingDate" placeholder="" class="filter-input" />
                            </th>
                            <th class="px-2 py-2">
                                <!--<input v-model="columnFilters.qty" placeholder="" class="filter-input" />-->
                            </th>
                            <th class="px-2 py-2">
                                <!--<input v-model="columnFilters.unit" placeholder="" class="filter-input" />-->
                            </th>
                            <th class="px-2 py-2">
                                <input v-model="columnFilters.keeperName" placeholder="" class="filter-input" />
                            </th>
                            <th class="px-2 py-2">
                                <input v-model="columnFilters.keeperDept" placeholder="" class="filter-input" />
                            </th>
                            <th class="px-2 py-2">
                                <input v-model="columnFilters.location" placeholder="" class="filter-input" />
                            </th>
                        </tr>
                    </thead>

                    <tbody class="bg-transparent divide-y divide-slate-200/50">
                        <tr v-if="filteredPurchases.length === 0">
                            <td colspan="12" class="px-6 py-16 text-center">
                                <div class="text-slate-700 font-bold mb-1">沒有符合篩選條件的資料</div>
                                <div class="text-sm text-slate-500">請直接清空上方欄位篩選條件。</div>
                            </td>
                        </tr>

                        <tr v-for="(item, index) in filteredPurchases"
                            :key="`${item.docNo}-${item.seq}-${index}`"
                            class="hover:bg-white/60 transition-colors">
                            <td class="px-2 py-3 text-center">
                                <input type="checkbox"
                                       v-model="item.selected"
                                       class="h-4 w-4">
                            </td>

                            <td class="table-td">
                                <div class="cell-main"
                                     :title="`${item.docTypeNo || ''}-${item.docNo || ''}-${item.seq || ''}`">
                                    {{ item.docNo }}
                                </div>
                                <div class="cell-sub">
                                    {{ item.docTypeNo }} / 序 {{ item.seq }}
                                </div>
                            </td>

                            <td class="table-td">
                                <div class="cell-text" :title="item.itemNo">{{ item.itemNo || '-' }}</div>
