<script setup>
    import { ref, computed, watch } from 'vue';
    import axios from 'axios';
    import AmsManagedAssetTransferPanel from '../components/assets/AmsManagedAssetTransferPanel.vue';

    const props = defineProps({
        currentUser: Object,
        isTransferBlockedToday: Boolean,
        blockReasonText: String,
        permissions: { type: Array, default: () => [] },
        myPermissions: { type: Array, default: () => [] }
    });

    const PERM = {
        AssetListView: 'Asset.List.View',
        TransferCreate: 'Transfer.Create'
    };

    const permSet = computed(() => new Set((props.permissions || []).map(String)));

    // const hasPerm = (key) => (props.myPermissions || []).includes(key)
    const hasPerm = (key) => {
        // 兼容：先用 permissions，沒有再用 myPermissions
        const list = (props.permissions?.length ? props.permissions : props.myPermissions) || [];
        return list.includes(key);
    };

    const emit = defineEmits(['openCreateTransfer']);

    const assets = ref([]);
    const isLoading = ref(false);
    const assetSearchQuery = ref('');
    const selectedAssets = ref([]);
    const currentPage = ref(1);
    const pageSize = 10;
    const API_BASE_URL = '/api';

    // === 新增：頁籤狀態 ===
    const activeTab = ref('erp-transfer');
    const amsSelectedAssets = ref([]);

    const handleAmsSelectionChanged = (items) => {
        amsSelectedAssets.value = items || [];
    };

    // ✅ 新增：系統參數 AssetNoPrefixes（全系統套用，由管理者在系統參數設定維護）
    const assetNoPrefixes = ref([]); // e.g. ["1656", "7788"]

    // ✅ 新增：讀取 public configs（只取我們需要的 prefixes）
    const fetchAssetNoPrefixes = async () => {
        try {
            const res = await axios.get(`${API_BASE_URL}/admin/config/public`);
            const data = res.data || {};
            const prefixes = data.AssetNoPrefixes || data.assetNoPrefixes || [];

            if (Array.isArray(prefixes)) {
                assetNoPrefixes.value = prefixes
                    .map(x => String(x || '').trim())
                    .filter(Boolean);
            } else {
                assetNoPrefixes.value = [];
            }
        } catch (e) {
            console.error('讀取 AssetNoPrefixes 失敗:', e);
            assetNoPrefixes.value = [];
        }
    };

    // Fetch Logic
    const fetchAssets = async () => {
        if (!props.currentUser) return;
        isLoading.value = true;
        try {
            const response = await axios.get(`${API_BASE_URL}/assets/my`);
            assets.value = (response.data || []).map(a => {
                const isBackendLocked = a.isLocked || a.IsLocked || false;
                const backendReason = (a.lockReason || a.LockReason) || (isBackendLocked ? '處理中' : '');

                return {
                    id: a.id || a.Id || a.assetNo || '',
                    name: a.name || a.Name || a.assetName || '',
                    qty: a.qty || a.Qty || 0,
                    unit: a.unit || a.Unit || '個',
                    company: a.company || a.Company || a.companyName || '未知',
                    departmentId: a.departmentId || a.DepartmentId,
                    departmentName: a.departmentName || a.DepartmentName,

                    _backendLocked: isBackendLocked,
                    _backendReason: backendReason,

                    isLocked: isBackendLocked || props.isTransferBlockedToday,
                    lockReason: isBackendLocked ? backendReason : ''
                };
            });

            if (props.isTransferBlockedToday) {
                selectedAssets.value = [];
            }

        } catch (error) {
            console.error('抓取資產失敗:', error);
            assets.value = [];
        } finally {
            isLoading.value = false;
        }
    };

    // ✅ 修改：先抓 prefixes，再抓 assets
    watch(() => props.currentUser, async (newVal) => {
        if (newVal) {
            await fetchAssetNoPrefixes();
            await fetchAssets();
        }
    }, { immediate: true });

    // 當全域鎖定狀態改變時，重新計算顯示邏輯
    watch(() => props.isTransferBlockedToday, (isGlobalBlocked) => {
        if (assets.value.length > 0) {
            assets.value = assets.value.map(a => ({
                ...a,
                isLocked: a._backendLocked || isGlobalBlocked,
                lockReason: a._backendLocked ? a._backendReason : ''
            }));

            if (isGlobalBlocked) {
                selectedAssets.value = [];
            }
        }
    });

    // 搜尋條件改變時回到第一頁
    watch(assetSearchQuery, () => {
        currentPage.value = 1;
    });

    // Filter & Pagination
    const filteredAssets = computed(() => {
        let list = assets.value || [];

        if (assetNoPrefixes.value && assetNoPrefixes.value.length > 0) {
            list = list.filter(a => {
                const assetNo = String(a.id || '');
                return assetNoPrefixes.value.some(p => assetNo.startsWith(p));
            });
        }

        if (!assetSearchQuery.value) return list;
        const q = assetSearchQuery.value.toLowerCase();
        return list.filter(a =>
            String(a.id || '').toLowerCase().includes(q) ||
            String(a.name || '').toLowerCase().includes(q)
        );
    });

    const paginatedAssets = computed(() => {
        const start = (currentPage.value - 1) * pageSize;
        const end = start + pageSize;
        return filteredAssets.value?.slice(start, end) || [];
    });

    const totalPages = computed(() => Math.ceil((filteredAssets.value?.length || 0) / pageSize));

    // Selection Logic
    const isAllSelected = computed(() => {
        const selectable = (filteredAssets.value || []).filter(a => !a.isLocked);
        return selectable.length > 0 && selectable.every(a => selectedAssets.value.includes(a.id));
    });

    const toggleSelection = (id) => {
        const idx = selectedAssets.value.indexOf(id);
        if (idx === -1) selectedAssets.value.push(id);
        else selectedAssets.value.splice(idx, 1);
    };

    const toggleAll = () => {
        const selectable = (filteredAssets.value || []).filter(a => !a.isLocked);
        if (selectable.length === 0) return;
        const allSelected = selectable.every(a => selectedAssets.value.includes(a.id));
        if (allSelected) {
            const idsToRemove = selectable.map(a => a.id);
            selectedAssets.value = selectedAssets.value.filter(id => !idsToRemove.includes(id));
        } else {
            const idsToAdd = selectable.map(a => a.id);
            idsToAdd.forEach(id => {
                if (!selectedAssets.value.includes(id)) selectedAssets.value.push(id);
            });
        }
    };

    const handleTransferClick = () => {
        emit('openCreateTransfer', filteredAssets.value.filter(a => selectedAssets.value.includes(a.id)));
    };

    const exportExcel = () => {
        const dataToExport = filteredAssets.value;
        if (!dataToExport || dataToExport.length === 0) return;
        let csvContent = "data:text/csv;charset=utf-8,\uFEFF資產編號,資產名稱,數量,單位,公司別\n";
        dataToExport.forEach(item => {
            const row = [`"${item.id}"`, `"${item.name}"`, item.qty, `"${item.unit}"`, `"${item.company}"`].join(",");
            csvContent += row + "\n";
        });
        const link = document.createElement("a");
        link.setAttribute("href", encodeURI(csvContent));
        link.setAttribute("download", "保管資產清單.csv");
        document.body.appendChild(link);
        link.click();
        document.body.removeChild(link);
    };

    // === 按鈕權限狀態 ===
    const canExport = computed(() => hasPerm(PERM.AssetListView));
    const canCreateTransfer = computed(() => hasPerm(PERM.TransferCreate));
</script>

<template>
    <div class="flex-1 flex flex-col h-full overflow-hidden">
        <!-- 頁籤 -->
        <div class="mb-4 bg-white/60 rounded-xl border border-slate-200 shadow-sm p-2 flex flex-wrap gap-2 shrink-0">
            <button @click="activeTab = 'erp-transfer'"
                    class="px-4 py-2 rounded-lg text-sm font-medium transition-all"
                    :class="activeTab === 'erp-transfer'
                        ? 'bg-blue-600 text-white shadow-md shadow-blue-300/40'
                        : 'bg-white/70 text-slate-600 hover:bg-white'">
                ERP 資產移轉
            </button>

            <button @click="activeTab = 'ams-transfer'"
                    class="px-4 py-2 rounded-lg text-sm font-medium transition-all"
                    :class="activeTab === 'ams-transfer'
                        ? 'bg-blue-600 text-white shadow-md shadow-blue-300/40'
                        : 'bg-white/70 text-slate-600 hover:bg-white'">
                AMS 列管資產移轉
                <span v-if="amsSelectedAssets.length > 0"
                      class="ml-2 bg-white/20 px-1.5 py-0.5 rounded-full text-xs font-bold min-w-[1.5rem] inline-block text-center">
                    {{ amsSelectedAssets.length }}
                </span>
            </button>
        </div>

        <!-- ERP 資產移轉：保留原本畫面 -->
        <div v-if="activeTab === 'erp-transfer'" class="flex-1 flex flex-col min-h-0 overflow-hidden">
            <!-- 說明區塊 -->
            <div class="mb-4 p-4 bg-blue-50/60 border border-blue-100 rounded-xl shadow-sm flex items-start gap-3 text-slate-600 shrink-0">
                <svg xmlns="http://www.w3.org/2000/svg" viewBox="0 0 20 20" fill="currentColor" class="w-5 h-5 text-blue-500 mt-0.5 shrink-0">
                    <path fill-rule="evenodd" d="M18 10a8 8 0 11-16 0 8 8 0 0116 0zm-7-4a1 1 0 11-2 0 1 1 0 012 0zM9 9a1 1 0 000 2v3a1 1 0 001 1h1a1 1 0 100-2v-3a1 1 0 00-1-1H9z" clip-rule="evenodd" />
                </svg>
                <div>
                    <h3 class="font-bold mb-1 text-blue-700">作業說明</h3>
                    <ul class="list-disc list-inside text-sm space-y-1 text-slate-600">
                        <li>人事異動時，請確認資產是否移交保管；資料若有誤請聯繫資訊或財務單位。</li>
                        <li>移轉單紀錄，可查詢自己發起移轉單及接收保管資產移轉單。</li>
                        <li>每月25~次月1日關帳期間，系統將鎖定資產移轉功能，期間請洽資訊、財務協助處理。</li>
                    </ul>
                </div>
            </div>

            <!-- 工具列 -->
            <div class="flex justify-end items-center mb-4 gap-3 shrink-0">
                <!-- 匯出：需要 Asset.List.View -->
                <button v-if="hasPerm('Asset.List.View')"
                        @click="exportExcel"
                        class="px-3 py-2 bg-white/60 hover:bg-white text-slate-700 rounded-lg shadow-sm border border-slate-200 transition-all flex items-center gap-2 hover:shadow-md">
                    <svg xmlns="http://www.w3.org/2000/svg" viewBox="0 0 20 20" fill="currentColor" class="w-5 h-5 text-green-600">
                        <path fill-rule="evenodd" d="M3 17a1 1 0 011-1h12a1 1 0 110 2H4a1 1 0 01-1-1zm3.293-7.707a1 1 0 011.414 0L9 10.586V3a1 1 0 112 0v7.586l1.293-1.293a1 1 0 111.414 1.414l-3 3a1 1 0 01-1.414 0l-3-3a1 1 0 010-1.414z" clip-rule="evenodd" />
                    </svg>
                    <span class="hidden sm:inline font-medium">匯出所有保管資產</span>
                </button>

                <!-- 資產移轉：需要 Transfer.Create -->
                <button @click="handleTransferClick"
                        :disabled="selectedAssets.length === 0 || !canCreateTransfer"
                        :title="!canCreateTransfer ? '你沒有「資產移轉」權限' : ''"
                        :class="(selectedAssets.length > 0 && canCreateTransfer)
                            ? 'bg-blue-700 hover:bg-blue-800 text-white shadow-blue-300/50'
                            : 'bg-slate-300 text-slate-500 cursor-not-allowed'"
                        class="px-4 py-2 rounded-lg shadow-md transition-all flex items-center gap-2 font-medium hover:shadow-lg transform active:scale-95">
                    <svg xmlns="http://www.w3.org/2000/svg" viewBox="0 0 24 24" fill="currentColor" class="w-4 h-4">
                        <path fill-rule="evenodd" d="M15.97 2.47a.75.75 0 011.06 0l4.5 4.5a.75.75 0 010 1.06l-4.5 4.5a.75.75 0 11-1.06-1.06l3.22-3.22H7.5a.75.75 0 010-1.5h11.69l-3.22-3.22a.75.75 0 010-1.06zm-7.94 9a.75.75 0 010 1.06l-3.22 3.22H16.5a.75.75 0 010 1.5H4.81l3.22 3.22a.75.75 0 11-1.06 1.06l-4.5-4.5a.75.75 0 010-1.06l4.5-4.5a.75.75 0 011.06 0z" clip-rule="evenodd" />
                    </svg>
                    <span>資產移轉</span>
                    <span v-if="selectedAssets.length > 0" class="bg-white/20 px-1.5 py-0.5 rounded-full text-xs font-bold min-w-[1.5rem] text-center">
                        {{ selectedAssets.length }}
                    </span>
                </button>

                <div class="relative w-64">
                    <input type="text" v-model="assetSearchQuery" placeholder="搜尋資產編號、名稱..."
                           class="w-full px-4 py-2 bg-white/60 border border-slate-200 rounded-lg focus:outline-none focus:ring-2 focus:ring-blue-500/20 focus:border-blue-500 transition-all shadow-sm" />
                </div>
            </div>

            <!-- 表格 -->
            <div class="flex-1 overflow-hidden flex flex-col">
                <div class="flex-1 overflow-auto rounded-xl border border-white/60 bg-white/40 shadow-inner">
                    <div v-if="isLoading" class="p-8 text-center text-slate-500">載入中...</div>
                    <table v-else class="w-full text-left border-collapse">
                        <thead class="sticky top-0 bg-white/90 backdrop-blur-md z-10 border-b border-slate-200/50">
                            <tr>
                                <th class="p-4 text-center">
                                    <input type="checkbox"
                                           :checked="isAllSelected"
                                           @change="toggleAll"
                                           class="w-4 h-4 rounded border-slate-300 text-blue-600 focus:ring-blue-500 cursor-pointer">
                                </th>
                                <th class="p-4 font-semibold text-slate-600 text-sm">資產編號</th>
                                <th class="p-4 font-semibold text-slate-600 text-sm">資產名稱</th>
                                <th class="p-4 font-semibold text-slate-600 text-sm">數量</th>
                                <th class="p-4 font-semibold text-slate-600 text-sm">單位</th>
                                <th class="p-4 font-semibold text-slate-600 text-sm">公司別</th>
                            </tr>
                        </thead>

                        <tbody class="divide-y divide-slate-200/50">
                            <tr v-if="!paginatedAssets || paginatedAssets.length === 0">
                                <td colspan="6" class="p-8 text-center text-slate-500">無符合條件資產</td>
                            </tr>

                            <tr v-for="asset in paginatedAssets" :key="asset.id"
                                class="transition-colors group"
                                :class="{
                                    'bg-blue-50/50': selectedAssets.includes(asset.id),
                                    'hover:bg-white/60 cursor-pointer': !asset.isLocked,
                                    'bg-gray-100/50 cursor-not-allowed opacity-60': asset.isLocked
                                }"
                                @click="!asset.isLocked && toggleSelection(asset.id)">
                                <td class="p-4 text-center" @click.stop>
                                    <input type="checkbox" :value="asset.id" v-model="selectedAssets"
                                           :disabled="asset.isLocked"
                                           class="w-4 h-4 rounded border-slate-300 text-blue-600 focus:ring-blue-500 cursor-pointer disabled:cursor-not-allowed disabled:bg-slate-200">
                                </td>
                                <td class="p-4 text-slate-700 font-mono text-sm font-medium" :class="{ 'group-hover:text-blue-600': !asset.isLocked }">
                                    {{ asset.id }}
                                </td>
                                <td class="p-4 text-slate-700 font-medium">
                                    {{ asset.name }}
                                    <span v-if="asset.isLocked && asset.lockReason"
                                          class="ml-2 px-1.5 py-0.5 rounded text-[10px] font-bold bg-red-100 text-red-500 border border-red-200 align-middle">
                                        {{ asset.lockReason }}
                                    </span>
                                </td>
                                <td class="p-4 text-slate-500">{{ asset.qty }}</td>
                                <td class="p-4 text-slate-500">{{ asset.unit }}</td>
                                <td class="p-4 text-slate-500">
                                    <span class="bg-slate-100 text-slate-600 px-2 py-0.5 rounded text-xs border border-slate-200">
                                        {{ asset.company }}
                                    </span>
                                </td>
                            </tr>
                        </tbody>
                    </table>
                </div>

                <!-- 分頁 -->
                <div class="flex justify-between items-center mt-4 pt-4 border-t border-slate-300/30 shrink-0">
                    <div class="text-xs text-slate-500">
                        顯示第 {{ filteredAssets.length === 0 ? 0 : (currentPage - 1) * pageSize + 1 }}
                        到 {{ Math.min(currentPage * pageSize, filteredAssets?.length || 0) }} 筆，
                        共 {{ filteredAssets?.length || 0 }} 筆
                    </div>
                    <div class="flex items-center gap-2">
                        <button @click="currentPage > 1 && currentPage--" :disabled="currentPage === 1"
                                class="p-1 rounded hover:bg-white/50 disabled:opacity-30 disabled:cursor-not-allowed text-slate-600">
                            <svg xmlns="http://www.w3.org/2000/svg" viewBox="0 0 20 20" fill="currentColor" class="w-5 h-5">
                                <path fill-rule="evenodd" d="M12.707 5.293a1 1 0 010 1.414L9.414 10l3.293 3.293a1 1 0 01-1.414 1.414l-4-4a1 1 0 010-1.414l4-4a1 1 0 011.414 0z" clip-rule="evenodd" />
                            </svg>
                        </button>
                        <div class="flex gap-1">
                            <button v-for="page in totalPages" :key="page" @click="currentPage = page"
                                    class="w-8 h-8 rounded text-sm font-medium transition-all"
                                    :class="currentPage === page ? 'bg-blue-600 text-white shadow-md' : 'hover:bg-white/50 text-slate-600'">
                                {{ page }}
                            </button>
                        </div>
                        <button @click="currentPage < totalPages && currentPage++" :disabled="currentPage === totalPages || totalPages === 0"
                                class="p-1 rounded hover:bg-white/50 disabled:opacity-30 disabled:cursor-not-allowed text-slate-600">
                            <svg xmlns="http://www.w3.org/2000/svg" viewBox="0 0 20 20" fill="currentColor" class="w-5 h-5">
                                <path fill-rule="evenodd" d="M7.293 14.707a1 1 0 010-1.414L10.586 10 7.293 6.707a1 1 0 011.414-1.414l4 4a1 1 0 010 1.414l-4 4a1 1 0 01-1.414 0z" clip-rule="evenodd" />
                            </svg>
                        </button>
                    </div>
                </div>
            </div>
        </div>

        <!-- AMS 列管資產移轉：第一階段只查詢與勾選 -->
        <AmsManagedAssetTransferPanel v-else-if="activeTab === 'ams-transfer'"
                                      class="flex-1 min-h-0 overflow-hidden"
                                      :current-user="currentUser"
                                      :permissions="permissions"
                                      :my-permissions="myPermissions"
                                      :is-transfer-blocked-today="isTransferBlockedToday"
                                      :block-reason-text="blockReasonText"
                                      @selection-changed="handleAmsSelectionChanged" />
    </div>
</template>
