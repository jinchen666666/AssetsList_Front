<script setup>
    import { ref, computed, onMounted, watch } from 'vue'
    import axios from 'axios'
    import AmsManagedAssetTransferModal from './AmsManagedAssetTransferModal.vue'

    const props = defineProps({
        currentUser: Object,
        isTransferBlockedToday: Boolean,
        blockReasonText: String,
        permissions: { type: Array, default: () => [] },
        myPermissions: { type: Array, default: () => [] }
    })

    const emit = defineEmits(['selectionChanged'])

    const API_BASE_URL = '/api'

    const keyword = ref('')
    const rows = ref([])
    const selectedIds = ref([])
    const isLoading = ref(false)
    const errorMessage = ref('')
    const successMessage = ref('')

    const page = ref(1)
    const pageSize = ref(10)
    const total = ref(0)

    const showTransferModal = ref(false)

    const totalPages = computed(() => {
        const size = Number(pageSize.value) || 10
        return Math.max(1, Math.ceil((Number(total.value) || 0) / size))
    })

    const selectedRows = computed(() =>
        rows.value.filter(x => selectedIds.value.includes(x.id))
    )

    const hasRows = computed(() => rows.value.length > 0)

    const isAllCurrentPageSelected = computed(() => {
        const selectable = rows.value.filter(x => !x.isLocked)
        return selectable.length > 0 && selectable.every(x => selectedIds.value.includes(x.id))
    })

    const canCreateTransfer = computed(() =>
        selectedIds.value.length > 0 && !props.isTransferBlockedToday
    )

    const currentUserId = computed(() => {
        const u = props.currentUser || {}
        return u.userId || u.UserId || u.employeeId || u.EmployeeId || u.id || u.Id || ''
    })

    const normalizeRow = (x) => ({
        id: Number(x.id ?? x.Id ?? 0),
        assetNo: x.assetNo || x.AssetNo || '',
        assetName: x.assetName || x.AssetName || '',
        assetSpec: x.assetSpec || x.AssetSpec || '',
        itemNo: x.itemNo || x.ItemNo || '',
        qty: x.qty ?? x.Qty ?? 0,
        unit: x.unit || x.Unit || '',
        company: x.company || x.Company || 'AMS',
        location: x.location || x.Location || '',
        keeperId: x.keeperId || x.KeeperId || '',
        keeperName: x.keeperName || x.KeeperName || '',
        keeperDept: x.keeperDept || x.KeeperDept || '',
        isLocked: x.isLocked || x.IsLocked || false,
        lockReason: x.lockReason || x.LockReason || ''
    })

    const fetchRows = async () => {
        isLoading.value = true
        errorMessage.value = ''

        try {
            const res = await axios.get(`${API_BASE_URL}/ams-asset-transfer/my-assets`, {
                params: {
                    keyword: keyword.value || undefined,
                    page: page.value,
                    pageSize: pageSize.value
                },
                headers: currentUserId.value
                    ? { 'X-User-Id': currentUserId.value }
                    : {}
            })

            const data = res.data || {}
            const rawItems = data.items || data.Items || []

            rows.value = Array.isArray(rawItems)
                ? rawItems.map(normalizeRow)
                : []

            total.value = Number(data.total ?? data.Total ?? rows.value.length) || 0

            selectedIds.value = selectedIds.value.filter(id =>
                rows.value.some(x => x.id === id && !x.isLocked)
            )
        } catch (e) {
            console.error(e)
            rows.value = []
            total.value = 0
            selectedIds.value = []
            errorMessage.value = e?.response?.data?.message || e?.response?.data?.Message || e?.message || 'AMS 列管資產查詢失敗'
        } finally {
            isLoading.value = false
        }
    }

    const search = async () => {
        page.value = 1
        selectedIds.value = []
        successMessage.value = ''
        await fetchRows()
    }

    const resetSearch = async () => {
        keyword.value = ''
        page.value = 1
        selectedIds.value = []
        successMessage.value = ''
        await fetchRows()
    }

    const toggleOne = (row) => {
        if (!row?.id || row.isLocked || props.isTransferBlockedToday) return

        if (selectedIds.value.includes(row.id)) {
            selectedIds.value = selectedIds.value.filter(x => x !== row.id)
        } else {
            selectedIds.value = [...selectedIds.value, row.id]
        }
    }

    const toggleCurrentPage = () => {
        if (props.isTransferBlockedToday) return

        const selectableIds = rows.value
            .filter(x => !x.isLocked)
            .map(x => x.id)
            .filter(Boolean)

        if (selectableIds.length === 0) return

        if (isAllCurrentPageSelected.value) {
            selectedIds.value = selectedIds.value.filter(id => !selectableIds.includes(id))
        } else {
            selectedIds.value = Array.from(new Set([...selectedIds.value, ...selectableIds]))
        }
    }

    const prevPage = async () => {
        if (page.value <= 1) return
        page.value -= 1
        await fetchRows()
    }

    const nextPage = async () => {
        if (page.value >= totalPages.value) return
        page.value += 1
        await fetchRows()
    }

    const openCreateTransfer = () => {
        if (!canCreateTransfer.value) return
        errorMessage.value = ''
        successMessage.value = ''
        showTransferModal.value = true
    }

    const handleSubmitted = async (result) => {
        const requestNo = result?.requestNo || result?.RequestNo || ''
        const sheetNo = result?.easyFlowSheetNo || result?.EasyFlowSheetNo || ''

        successMessage.value = `AMS 列管資產移轉申請已送簽${requestNo ? `，申請單號：${requestNo}` : ''}${sheetNo ? `，簽核單號：${sheetNo}` : ''}`

        selectedIds.value = []
        await fetchRows()
    }

    watch(selectedRows, (v) => {
        emit('selectionChanged', v)
    })

    watch(() => props.isTransferBlockedToday, (blocked) => {
        if (blocked) {
            selectedIds.value = []
        }
    })

    onMounted(fetchRows)
</script>

<template>
    <div class="flex-1 flex flex-col h-full overflow-hidden">
        <div class="mb-4 p-4 bg-blue-50/60 border border-blue-100 rounded-xl shadow-sm flex items-start gap-3 text-slate-600 shrink-0">
            <svg xmlns="http://www.w3.org/2000/svg"
                 viewBox="0 0 20 20"
                 fill="currentColor"
                 class="w-5 h-5 text-blue-500 mt-0.5 shrink-0">
                <path fill-rule="evenodd"
                      d="M18 10a8 8 0 11-16 0 8 8 0 0116 0zm-7-4a1 1 0 11-2 0 1 1 0 012 0zM9 9a1 1 0 000 2v3a1 1 0 001 1h1a1 1 0 100-2v-3a1 1 0 00-1-1H9z"
                      clip-rule="evenodd" />
            </svg>
            <div>
                <h3 class="font-bold mb-1 text-blue-700">作業說明</h3>
                <ul class="list-disc list-inside text-sm space-y-1 text-slate-600">
                    <li>此頁顯示您目前保管的 AMS 列管資產，請勾選需移轉的資產後建立移轉申請。</li>
                    <li>本頁僅能查詢自己的保管資產；若需協助他人移轉，後續將由總務 / IT 專用功能處理。</li>
                    <li>每月25~次月1日關帳期間，系統將鎖定資產移轉功能，期間請洽資訊、財務協助處理。</li>
                </ul>
            </div>
        </div>

        <div v-if="isTransferBlockedToday"
             class="mb-4 bg-amber-50/80 border border-amber-200 text-amber-700 rounded-xl px-4 py-3 text-sm shrink-0">
            目前為移轉關帳期間，暫停建立移轉申請。{{ blockReasonText || '' }}
        </div>

        <div v-if="successMessage"
             class="mb-4 bg-green-50/80 border border-green-200 text-green-700 rounded-xl px-4 py-3 text-sm shrink-0">
            {{ successMessage }}
        </div>

        <div class="flex justify-end items-center mb-4 gap-3 shrink-0">
            <button @click="openCreateTransfer"
                    :disabled="!canCreateTransfer"
                    :class="canCreateTransfer
                        ? 'bg-blue-700 hover:bg-blue-800 text-white shadow-blue-300/50'
                        : 'bg-slate-300 text-slate-500 cursor-not-allowed'"
                    class="px-4 py-2 rounded-lg shadow-md transition-all flex items-center gap-2 font-medium hover:shadow-lg transform active:scale-95">
                <svg xmlns="http://www.w3.org/2000/svg"
                     viewBox="0 0 24 24"
                     fill="currentColor"
                     class="w-4 h-4">
                    <path fill-rule="evenodd"
                          d="M15.97 2.47a.75.75 0 011.06 0l4.5 4.5a.75.75 0 010 1.06l-4.5 4.5a.75.75 0 11-1.06-1.06l3.22-3.22H7.5a.75.75 0 010-1.5h11.69l-3.22-3.22a.75.75 0 010-1.06zm-7.94 9a.75.75 0 010 1.06l-3.22 3.22H16.5a.75.75 0 010 1.5H4.81l3.22 3.22a.75.75 0 11-1.06 1.06l-4.5-4.5a.75.75 0 010-1.06l4.5-4.5a.75.75 0 011.06 0z"
                          clip-rule="evenodd" />
                </svg>
                <span>資產移轉</span>
                <span v-if="selectedIds.length > 0"
                      class="bg-white/20 px-1.5 py-0.5 rounded-full text-xs font-bold min-w-[1.5rem] text-center">
                    {{ selectedIds.length }}
                </span>
            </button>

            <button @click="resetSearch"
                    class="px-3 py-2 bg-white/60 hover:bg-white text-slate-700 rounded-lg shadow-sm border border-slate-200 transition-all flex items-center gap-2 hover:shadow-md">
                <span class="font-medium">清除</span>
            </button>

            <button @click="search"
                    :disabled="isLoading"
                    class="px-3 py-2 bg-blue-600 hover:bg-blue-700 text-white rounded-lg shadow-sm border border-blue-500 transition-all flex items-center gap-2 hover:shadow-md disabled:opacity-50">
                <span class="font-medium">{{ isLoading ? '查詢中...' : '查詢' }}</span>
            </button>

            <div class="relative w-64">
                <input type="text"
                       v-model="keyword"
                       @keyup.enter="search"
                       placeholder="搜尋資產編號、名稱..."
                       class="w-full px-4 py-2 bg-white/60 border border-slate-200 rounded-lg focus:outline-none focus:ring-2 focus:ring-blue-500/20 focus:border-blue-500 transition-all shadow-sm" />
            </div>
        </div>

        <div v-if="errorMessage"
             class="mb-4 bg-red-50/80 border border-red-200 text-red-700 rounded-xl px-4 py-3 text-sm shrink-0">
            {{ errorMessage }}
        </div>

        <div class="flex-1 overflow-hidden flex flex-col">
            <div class="flex-1 overflow-auto rounded-xl border border-white/60 bg-white/40 shadow-inner">
                <div v-if="isLoading" class="p-8 text-center text-slate-500">載入中...</div>

                <table v-else class="w-full text-left border-collapse">
                    <thead class="sticky top-0 bg-white/90 backdrop-blur-md z-10 border-b border-slate-200/50">
                        <tr>
                            <th class="p-4 text-center">
                                <input type="checkbox"
                                       :checked="isAllCurrentPageSelected"
                                       :disabled="!hasRows || isTransferBlockedToday"
                                       @change="toggleCurrentPage"
                                       class="w-4 h-4 rounded border-slate-300 text-blue-600 focus:ring-blue-500 cursor-pointer disabled:cursor-not-allowed disabled:bg-slate-200">
                            </th>
                            <th class="p-4 font-semibold text-slate-600 text-sm">資產編號</th>
                            <th class="p-4 font-semibold text-slate-600 text-sm">資產名稱</th>
                            <th class="p-4 font-semibold text-slate-600 text-sm">數量</th>
                            <th class="p-4 font-semibold text-slate-600 text-sm">單位</th>
                            <th class="p-4 font-semibold text-slate-600 text-sm">放置地點</th>
                        </tr>
                    </thead>

                    <tbody class="divide-y divide-slate-200/50">
                        <tr v-if="!rows || rows.length === 0">
                            <td colspan="6" class="p-8 text-center text-slate-500">
                                無符合條件資產
                            </td>
                        </tr>

                        <tr v-for="asset in rows"
                            :key="asset.id"
                            class="transition-colors group"
                            :class="{
                                'bg-blue-50/50': selectedIds.includes(asset.id),
                                'hover:bg-white/60 cursor-pointer': !asset.isLocked && !isTransferBlockedToday,
                                'bg-gray-100/50 cursor-not-allowed opacity-60': asset.isLocked || isTransferBlockedToday
                            }"
                            @click="!asset.isLocked && !isTransferBlockedToday && toggleOne(asset)">
                            <td class="p-4 text-center" @click.stop>
                                <input type="checkbox"
                                       :value="asset.id"
                                       :checked="selectedIds.includes(asset.id)"
                                       :disabled="asset.isLocked || isTransferBlockedToday"
                                       @change="toggleOne(asset)"
                                       class="w-4 h-4 rounded border-slate-300 text-blue-600 focus:ring-blue-500 cursor-pointer disabled:cursor-not-allowed disabled:bg-slate-200">
                            </td>

                            <td class="p-4 text-slate-700 font-mono text-sm font-medium"
                                :class="{ 'group-hover:text-blue-600': !asset.isLocked && !isTransferBlockedToday }">
                                {{ asset.assetNo }}
                                <span v-if="asset.isLocked && asset.lockReason"
                                      class="ml-2 px-1.5 py-0.5 rounded text-[10px] font-bold bg-red-100 text-red-500 border border-red-200 align-middle">
                                    {{ asset.lockReason }}
                                </span>
                            </td>

                            <td class="p-4 text-slate-700 font-medium">
                                {{ asset.assetName }}
                            </td>

                            <td class="p-4 text-slate-500">
                                {{ asset.qty }}
                            </td>

                            <td class="p-4 text-slate-500">
                                {{ asset.unit || '-' }}
                            </td>

                            <td class="p-4 text-slate-500 max-w-[220px] truncate"
                                :title="asset.location">
                                {{ asset.location || '-' }}
                            </td>
                        </tr>
                    </tbody>
                </table>
            </div>

            <div class="flex justify-between items-center mt-4 pt-4 border-t border-slate-300/30 shrink-0">
                <div class="text-xs text-slate-500">
                    顯示第 {{ total === 0 ? 0 : (page - 1) * pageSize + 1 }}
                    到 {{ Math.min(page * pageSize, total) }} 筆，
                    共 {{ total }} 筆
                </div>

                <div class="flex items-center gap-2">
                    <button @click="prevPage"
                            :disabled="page <= 1 || isLoading"
                            class="p-1 rounded hover:bg-white/50 disabled:opacity-30 disabled:cursor-not-allowed text-slate-600">
                        上一頁
                    </button>

                    <div class="text-sm text-slate-600 px-2">
                        第 {{ page }} / {{ totalPages }} 頁
                    </div>

                    <button @click="nextPage"
                            :disabled="page >= totalPages || isLoading"
                            class="p-1 rounded hover:bg-white/50 disabled:opacity-30 disabled:cursor-not-allowed text-slate-600">
                        下一頁
                    </button>
                </div>
            </div>
        </div>

        <AmsManagedAssetTransferModal :show="showTransferModal"
                                      :assets="selectedRows"
                                      :current-user="currentUser"
                                      @close="showTransferModal = false"
                                      @submitted="handleSubmitted" />
    </div>
</template>
