<script setup>
    import { ref, computed, watch } from 'vue';

    const props = defineProps({
        systemConfigs: {
            type: Array,
            default: () => []
        },
        canAdmin: Boolean
    });

    const emit = defineEmits(['save-batch']);

    const ASSET_CODE_DEFAULT_GENERAL_AFFAIRS = 'AssetCode.Default.GeneralAffairs';
    const ASSET_CODE_DEFAULT_ASSET_CATEGORY = 'AssetCode.Default.AssetCategory';
    const ASSET_CODE_DEFAULT_ITEM_CATEGORY = 'AssetCode.Default.ItemCategory';

    const ASSET_CODE_GENERAL_AFFAIRS_OPTIONS = 'AssetCode.GeneralAffairsOptions';
    const ASSET_CODE_ASSET_CATEGORY_OPTIONS = 'AssetCode.AssetCategoryOptions';
    const ASSET_CODE_ITEM_CATEGORY_OPTIONS = 'AssetCode.ItemCategoryOptions';

    const assetCodeDefaultGeneralAffairs = ref('99');
    const assetCodeDefaultAssetCategory = ref('04');
    const assetCodeDefaultItemCategory = ref('TB');

    const generalAffairsRows = ref([]);
    const assetCategoryRows = ref([]);
    const itemCategoryRows = ref([]);

    const newGeneralAffairsCode = ref('');
    const newGeneralAffairsName = ref('');
    const newAssetCategoryCode = ref('');
    const newAssetCategoryName = ref('');
    const newItemCategoryCode = ref('');
    const newItemCategoryName = ref('');

    const infoMessage = ref('');
    const isSavingAssetCode = ref(false);

    const getKey = (c) => c.configKey || c.ConfigKey || '';
    const getValue = (c) => (c.configValue ?? c.ConfigValue ?? '');

    const getConfigByKey = (key) => (props.systemConfigs || []).find(x => getKey(x) === key);

    const parseCodeOptions = (raw) => {
        return String(raw || '')
            .split(/\r?\n/g)
            .map(x => x.trim())
            .filter(Boolean)
            .map(line => {
                const [code, name] = line.split('|', 2);
                return {
                    code: (code || '').trim().toUpperCase(),
                    name: (name || '').trim()
                };
            })
            .filter(x => x.code);
    };

    const buildCodeOptions = (rows) => {
        const map = new Map();
        (rows || []).forEach(r => {
            const code = String(r.code || '').trim().toUpperCase();
            const name = String(r.name || '').trim();
            if (!code) return;
            map.set(code, name);
        });

        return Array.from(map.entries())
            .map(([code, name]) => `${code}|${name}`)
            .join('\n');
    };

    const syncFromConfigs = () => {
        generalAffairsRows.value = parseCodeOptions(getValue(getConfigByKey(ASSET_CODE_GENERAL_AFFAIRS_OPTIONS) || {}));
        assetCategoryRows.value = parseCodeOptions(getValue(getConfigByKey(ASSET_CODE_ASSET_CATEGORY_OPTIONS) || {}));
        itemCategoryRows.value = parseCodeOptions(getValue(getConfigByKey(ASSET_CODE_ITEM_CATEGORY_OPTIONS) || {}));

        assetCodeDefaultGeneralAffairs.value = getValue(getConfigByKey(ASSET_CODE_DEFAULT_GENERAL_AFFAIRS) || {}) || '99';
        assetCodeDefaultAssetCategory.value = getValue(getConfigByKey(ASSET_CODE_DEFAULT_ASSET_CATEGORY) || {}) || '04';
        assetCodeDefaultItemCategory.value = getValue(getConfigByKey(ASSET_CODE_DEFAULT_ITEM_CATEGORY) || {}) || 'TB';

        if (!assetCodeDefaultGeneralAffairs.value && generalAffairsRows.value.length > 0) {
            assetCodeDefaultGeneralAffairs.value = generalAffairsRows.value[0].code;
        }
        if (!assetCodeDefaultAssetCategory.value && assetCategoryRows.value.length > 0) {
            assetCodeDefaultAssetCategory.value = assetCategoryRows.value[0].code;
        }
        if (!assetCodeDefaultItemCategory.value && itemCategoryRows.value.length > 0) {
            assetCodeDefaultItemCategory.value = itemCategoryRows.value[0].code;
        }
    };

    const addRow = (rowsRef, code, name, codeLength, codeLabel, resetFn) => {
        if (!props.canAdmin) return;

        const normalizedCode = String(code || '').trim().toUpperCase();
        const normalizedName = String(name || '').trim();

        if (!normalizedCode) {
            infoMessage.value = `請輸入${codeLabel}代碼`;
            return;
        }

        if (normalizedCode.length !== codeLength) {
            infoMessage.value = `${codeLabel}代碼必須為 ${codeLength} 碼`;
            return;
        }

        const exists = rowsRef.value.some(
            x => String(x.code || '').trim().toUpperCase() === normalizedCode
        );

        if (exists) {
            infoMessage.value = `${codeLabel}代碼已存在`;
            return;
        }

        rowsRef.value = [
            ...rowsRef.value,
            { code: normalizedCode, name: normalizedName }
        ];

        resetFn();
        infoMessage.value = `${codeLabel}已加入下方列表，記得再按「儲存資產編碼設定」`;
    };

    const addGeneralAffairsRow = () => {
        addRow(
            generalAffairsRows,
            newGeneralAffairsCode.value,
            newGeneralAffairsName.value,
            2,
            '總務識別碼',
            () => {
                newGeneralAffairsCode.value = '';
                newGeneralAffairsName.value = '';
            }
        );
    };

    const addAssetCategoryRow = () => {
        addRow(
            assetCategoryRows,
            newAssetCategoryCode.value,
            newAssetCategoryName.value,
            2,
            '資產類別',
            () => {
                newAssetCategoryCode.value = '';
                newAssetCategoryName.value = '';
            }
        );
    };

    const addItemCategoryRow = () => {
        addRow(
            itemCategoryRows,
            newItemCategoryCode.value,
            newItemCategoryName.value,
            2,
            '品項類別',
            () => {
                newItemCategoryCode.value = '';
                newItemCategoryName.value = '';
            }
        );
    };

    const removeGeneralAffairsRow = (idx) => {
        if (!props.canAdmin) return;
        generalAffairsRows.value = generalAffairsRows.value.filter((_, i) => i !== idx);
        infoMessage.value = '已從下方列表移除，記得再按「儲存資產編碼設定」';
    };

    const removeAssetCategoryRow = (idx) => {
        if (!props.canAdmin) return;
        assetCategoryRows.value = assetCategoryRows.value.filter((_, i) => i !== idx);
        infoMessage.value = '已從下方列表移除，記得再按「儲存資產編碼設定」';
    };

    const removeItemCategoryRow = (idx) => {
        if (!props.canAdmin) return;
        itemCategoryRows.value = itemCategoryRows.value.filter((_, i) => i !== idx);
        infoMessage.value = '已從下方列表移除，記得再按「儲存資產編碼設定」';
    };

    const generalAffairsSelectOptions = computed(() =>
        Array.isArray(generalAffairsRows.value)
            ? generalAffairsRows.value.map(x => ({
                value: x.code,
                label: x.name ? `${x.code} - ${x.name}` : x.code
            }))
            : []
    );

    const assetCategorySelectOptions = computed(() =>
        Array.isArray(assetCategoryRows.value)
            ? assetCategoryRows.value.map(x => ({
                value: x.code,
                label: x.name ? `${x.code} - ${x.name}` : x.code
            }))
            : []
    );

    const itemCategorySelectOptions = computed(() =>
        Array.isArray(itemCategoryRows.value)
            ? itemCategoryRows.value.map(x => ({
                value: x.code,
                label: x.name ? `${x.code} - ${x.name}` : x.code
            }))
            : []
    );

    const save = () => {
        isSavingAssetCode.value = true;

        const payloads = [
            {
                ConfigKey: ASSET_CODE_GENERAL_AFFAIRS_OPTIONS,
                ConfigValue: buildCodeOptions(generalAffairsRows.value),
                Description: '格式 代碼|中文名稱，例如 99|總務'
            },
            {
                ConfigKey: ASSET_CODE_ASSET_CATEGORY_OPTIONS,
                ConfigValue: buildCodeOptions(assetCategoryRows.value),
                Description: '格式 代碼|中文名稱，例如 04|家具'
            },
            {
                ConfigKey: ASSET_CODE_ITEM_CATEGORY_OPTIONS,
                ConfigValue: buildCodeOptions(itemCategoryRows.value),
                Description: '格式 代碼|中文名稱，例如 TB|桌子'
            },
            {
                ConfigKey: ASSET_CODE_DEFAULT_GENERAL_AFFAIRS,
                ConfigValue: String(assetCodeDefaultGeneralAffairs.value || '').trim(),
                Description: '資產編號預設總務識別碼'
            },
            {
                ConfigKey: ASSET_CODE_DEFAULT_ASSET_CATEGORY,
                ConfigValue: String(assetCodeDefaultAssetCategory.value || '').trim(),
                Description: '資產編號預設資產類別碼'
            },
            {
                ConfigKey: ASSET_CODE_DEFAULT_ITEM_CATEGORY,
                ConfigValue: String(assetCodeDefaultItemCategory.value || '').trim(),
                Description: '資產編號預設品項類別碼'
            }
        ];

        emit('save-batch', payloads, () => {
            isSavingAssetCode.value = false;
            infoMessage.value = '';
        });
    };

    watch(
        () => props.systemConfigs,
        () => {
            syncFromConfigs();
        },
        { immediate: true }
    );
</script>

<template>
    <div class="bg-white/60 p-5 rounded-xl border border-slate-200 shadow-sm transition-all hover:shadow-md flex flex-col gap-6">
        <div class="flex flex-col gap-2 min-w-0">
            <h4 class="text-base font-bold text-slate-800 break-words">資產編碼設定</h4>
            <div>
                <span class="inline-flex items-center px-2 py-1 rounded-md text-xs font-mono font-bold bg-purple-50 text-purple-700 border border-purple-200/60 shadow-sm">
                    AssetCode.*
                </span>
            </div>
            <div class="text-sm text-slate-500 mt-2 bg-white/50 p-2.5 rounded-lg border border-slate-100">
                可維護資產編碼代碼、中文名稱與預設值。<br>
                編號格式：<code class="bg-slate-200 text-slate-800 px-1.5 py-0.5 rounded font-mono font-bold text-xs mx-0.5">總務識別碼2碼 + 資產類別2碼 + 品項類別2碼 + YYMM + 4碼流水號</code>
            </div>
            <div v-if="infoMessage" class="text-sm text-amber-700 bg-amber-50 border border-amber-200 rounded-lg px-3 py-2">
                {{ infoMessage }}
            </div>
        </div>

        <div class="border border-slate-200 rounded-xl p-4 bg-white/40">
            <div class="flex items-center justify-between mb-3">
                <h5 class="font-bold text-slate-700">總務識別碼對照表</h5>
            </div>

            <div class="flex flex-col lg:flex-row gap-2 lg:items-center mb-3">
                <input v-model="newGeneralAffairsCode" maxlength="2" placeholder="代碼，例如 99"
                       class="w-full lg:w-40 px-4 py-2.5 bg-white border border-slate-300 rounded-lg font-mono text-sm focus:ring-2 focus:ring-purple-500/20 focus:border-purple-500 outline-none transition-all shadow-sm" />
                <input v-model="newGeneralAffairsName" placeholder="中文名稱，例如 總務"
                       class="flex-1 px-4 py-2.5 bg-white border border-slate-300 rounded-lg text-sm focus:ring-2 focus:ring-purple-500/20 focus:border-purple-500 outline-none transition-all shadow-sm" />
                <button @click="addGeneralAffairsRow"
                        class="px-4 py-2.5 bg-white border border-slate-200 rounded-lg hover:bg-slate-50 text-slate-700 font-medium shadow-sm transition-all active:scale-95 text-sm whitespace-nowrap">
                    新增
                </button>
            </div>

            <div class="border border-slate-200 rounded-xl overflow-hidden bg-white/50">
                <table class="w-full text-sm">
                    <thead class="bg-slate-50 text-slate-500 font-medium border-b border-slate-200">
                        <tr>
                            <th class="p-3 w-32">代碼</th>
                            <th class="p-3">中文</th>
                            <th class="p-3 w-20 text-center">操作</th>
                        </tr>
                    </thead>
                    <tbody class="divide-y divide-slate-100">
                        <tr v-for="(r, i) in generalAffairsRows" :key="`${r.code}-${i}`">
                            <td class="p-3">
                                <input v-model="r.code" maxlength="2"
                                       class="w-full px-3 py-2 bg-white border border-slate-300 rounded-lg font-mono text-sm focus:ring-2 focus:ring-purple-500/20 focus:border-purple-500 outline-none transition-all shadow-sm" />
                            </td>
                            <td class="p-3">
                                <input v-model="r.name"
                                       class="w-full px-3 py-2 bg-white border border-slate-300 rounded-lg text-sm focus:ring-2 focus:ring-purple-500/20 focus:border-purple-500 outline-none transition-all shadow-sm" />
                            </td>
                            <td class="p-3 text-center">
                                <button @click="removeGeneralAffairsRow(i)" class="text-red-600 hover:underline text-sm">刪除</button>
                            </td>
                        </tr>
                        <tr v-if="generalAffairsRows.length === 0">
                            <td colspan="3" class="p-3 text-slate-400">尚未設定總務識別碼</td>
                        </tr>
                    </tbody>
                </table>
            </div>
        </div>

        <div class="border border-slate-200 rounded-xl p-4 bg-white/40">
            <div class="flex items-center justify-between mb-3">
                <h5 class="font-bold text-slate-700">資產類別對照表</h5>
            </div>

            <div class="flex flex-col lg:flex-row gap-2 lg:items-center mb-3">
                <input v-model="newAssetCategoryCode" maxlength="2" placeholder="代碼，例如 04"
                       class="w-full lg:w-40 px-4 py-2.5 bg-white border border-slate-300 rounded-lg font-mono text-sm focus:ring-2 focus:ring-purple-500/20 focus:border-purple-500 outline-none transition-all shadow-sm" />
                <input v-model="newAssetCategoryName" placeholder="中文名稱，例如 家具"
                       class="flex-1 px-4 py-2.5 bg-white border border-slate-300 rounded-lg text-sm focus:ring-2 focus:ring-purple-500/20 focus:border-purple-500 outline-none transition-all shadow-sm" />
                <button @click="addAssetCategoryRow"
                        class="px-4 py-2.5 bg-white border border-slate-200 rounded-lg hover:bg-slate-50 text-slate-700 font-medium shadow-sm transition-all active:scale-95 text-sm whitespace-nowrap">
                    新增
                </button>
            </div>

            <div class="border border-slate-200 rounded-xl overflow-hidden bg-white/50">
                <table class="w-full text-sm">
                    <thead class="bg-slate-50 text-slate-500 font-medium border-b border-slate-200">
                        <tr>
                            <th class="p-3 w-32">代碼</th>
                            <th class="p-3">中文</th>
                            <th class="p-3 w-20 text-center">操作</th>
                        </tr>
                    </thead>
                    <tbody class="divide-y divide-slate-100">
                        <tr v-for="(r, i) in assetCategoryRows" :key="`${r.code}-${i}`">
                            <td class="p-3">
                                <input v-model="r.code" maxlength="2"
                                       class="w-full px-3 py-2 bg-white border border-slate-300 rounded-lg font-mono text-sm focus:ring-2 focus:ring-purple-500/20 focus:border-purple-500 outline-none transition-all shadow-sm" />
                            </td>
                            <td class="p-3">
                                <input v-model="r.name"
                                       class="w-full px-3 py-2 bg-white border border-slate-300 rounded-lg text-sm focus:ring-2 focus:ring-purple-500/20 focus:border-purple-500 outline-none transition-all shadow-sm" />
                            </td>
                            <td class="p-3 text-center">
                                <button @click="removeAssetCategoryRow(i)" class="text-red-600 hover:underline text-sm">刪除</button>
                            </td>
                        </tr>
                        <tr v-if="assetCategoryRows.length === 0">
                            <td colspan="3" class="p-3 text-slate-400">尚未設定資產類別</td>
                        </tr>
                    </tbody>
                </table>
            </div>
        </div>

        <div class="border border-slate-200 rounded-xl p-4 bg-white/40">
            <div class="flex items-center justify-between mb-3">
                <h5 class="font-bold text-slate-700">品項類別對照表</h5>
            </div>

            <div class="flex flex-col lg:flex-row gap-2 lg:items-center mb-3">
                <input v-model="newItemCategoryCode" maxlength="2" placeholder="代碼，例如 TB"
                       class="w-full lg:w-40 px-4 py-2.5 bg-white border border-slate-300 rounded-lg font-mono text-sm focus:ring-2 focus:ring-purple-500/20 focus:border-purple-500 outline-none transition-all shadow-sm" />
                <input v-model="newItemCategoryName" placeholder="中文名稱，例如 桌子"
                       class="flex-1 px-4 py-2.5 bg-white border border-slate-300 rounded-lg text-sm focus:ring-2 focus:ring-purple-500/20 focus:border-purple-500 outline-none transition-all shadow-sm" />
                <button @click="addItemCategoryRow"
                        class="px-4 py-2.5 bg-white border border-slate-200 rounded-lg hover:bg-slate-50 text-slate-700 font-medium shadow-sm transition-all active:scale-95 text-sm whitespace-nowrap">
                    新增
                </button>
            </div>

            <div class="border border-slate-200 rounded-xl overflow-hidden bg-white/50">
                <table class="w-full text-sm">
                    <thead class="bg-slate-50 text-slate-500 font-medium border-b border-slate-200">
                        <tr>
                            <th class="p-3 w-32">代碼</th>
                            <th class="p-3">中文</th>
                            <th class="p-3 w-20 text-center">操作</th>
                        </tr>
                    </thead>
                    <tbody class="divide-y divide-slate-100">
                        <tr v-for="(r, i) in itemCategoryRows" :key="`${r.code}-${i}`">
                            <td class="p-3">
                                <input v-model="r.code" maxlength="2"
                                       class="w-full px-3 py-2 bg-white border border-slate-300 rounded-lg font-mono text-sm focus:ring-2 focus:ring-purple-500/20 focus:border-purple-500 outline-none transition-all shadow-sm" />
                            </td>
                            <td class="p-3">
                                <input v-model="r.name"
                                       class="w-full px-3 py-2 bg-white border border-slate-300 rounded-lg text-sm focus:ring-2 focus:ring-purple-500/20 focus:border-purple-500 outline-none transition-all shadow-sm" />
                            </td>
                            <td class="p-3 text-center">
                                <button @click="removeItemCategoryRow(i)" class="text-red-600 hover:underline text-sm">刪除</button>
                            </td>
                        </tr>
                        <tr v-if="itemCategoryRows.length === 0">
                            <td colspan="3" class="p-3 text-slate-400">尚未設定品項類別</td>
                        </tr>
                    </tbody>
                </table>
            </div>
        </div>

        <div class="border border-slate-200 rounded-xl p-4 bg-white/40">
            <h5 class="font-bold text-slate-700 mb-3">預設值設定</h5>

            <div class="grid grid-cols-1 md:grid-cols-3 gap-4">
                <div>
                    <label class="block text-sm font-medium text-slate-700 mb-2">預設總務識別碼</label>
                    <select v-model="assetCodeDefaultGeneralAffairs"
                            class="w-full px-4 py-2.5 bg-white border border-slate-300 rounded-lg text-sm focus:ring-2 focus:ring-purple-500/20 focus:border-purple-500 outline-none transition-all shadow-sm">
                        <option v-for="opt in generalAffairsSelectOptions" :key="opt.value" :value="opt.value">
                            {{ opt.label }}
                        </option>
                    </select>
                </div>

                <div>
                    <label class="block text-sm font-medium text-slate-700 mb-2">預設資產類別碼</label>
                    <select v-model="assetCodeDefaultAssetCategory"
                            class="w-full px-4 py-2.5 bg-white border border-slate-300 rounded-lg text-sm focus:ring-2 focus:ring-purple-500/20 focus:border-purple-500 outline-none transition-all shadow-sm">
                        <option v-for="opt in assetCategorySelectOptions" :key="opt.value" :value="opt.value">
                            {{ opt.label }}
                        </option>
                    </select>
                </div>

                <div>
                    <label class="block text-sm font-medium text-slate-700 mb-2">預設品項類別碼</label>
                    <select v-model="assetCodeDefaultItemCategory"
                            class="w-full px-4 py-2.5 bg-white border border-slate-300 rounded-lg text-sm focus:ring-2 focus:ring-purple-500/20 focus:border-purple-500 outline-none transition-all shadow-sm">
                        <option v-for="opt in itemCategorySelectOptions" :key="opt.value" :value="opt.value">
                            {{ opt.label }}
                        </option>
                    </select>
                </div>
            </div>
        </div>

        <div class="flex justify-end">
            <button @click="save"
                    :disabled="isSavingAssetCode"
                    class="px-6 py-2.5 bg-purple-600 hover:bg-purple-700 text-white shadow-md shadow-purple-300/50 hover:shadow-lg rounded-lg transition-all transform active:scale-95 flex items-center gap-2 font-medium text-sm whitespace-nowrap disabled:opacity-50">
                {{ isSavingAssetCode ? '儲存中...' : '儲存資產編碼設定' }}
            </button>
        </div>
    </div>
</template>
