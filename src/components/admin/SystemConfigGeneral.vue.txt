<script setup>
import { ref, computed, watch } from 'vue';

const props = defineProps({
    systemConfigs: {
        type: Array,
        default: () => []
    },
    canAdmin: Boolean
});

const emit = defineEmits(['save-config']);

const prefixInput = ref('');

const getKey = (c) => c.configKey || c.ConfigKey || '';
const getValue = (c) => (c.configValue ?? c.ConfigValue ?? '');
const getDesc = (c) => c.description || c.Description || '';
const setValue = (c, val) => { c.configValue = val; };

const filteredConfigs = computed(() =>
    (props.systemConfigs || []).filter(c =>
        ['Transfer_BlockPeriod', 'AssetNo_Prefixes'].includes(getKey(c))
    )
);

const parsePrefixes = (raw) => {
    return String(raw || '')
        .split(',')
        .map(x => x.trim())
        .filter(Boolean)
        .filter((v, i, arr) => arr.indexOf(v) === i);
};

const buildPrefixValue = (arr) => arr.join(',');

const addPrefixToConfig = (config) => {
    if (!props.canAdmin) return;

    const p = String(prefixInput.value || '').trim();
    if (!p || /\s/.test(p)) return;

    const current = parsePrefixes(getValue(config));
    if (!current.includes(p)) {
        current.push(p);
        setValue(config, buildPrefixValue(current));
    }
    prefixInput.value = '';
};

const removePrefixFromConfig = (config, p) => {
    if (!props.canAdmin) return;
    const current = parsePrefixes(getValue(config)).filter(x => x !== p);
    setValue(config, buildPrefixValue(current));
};

const save = (config) => {
    emit('save-config', {
        ConfigKey: getKey(config),
        ConfigValue: getValue(config),
        Description: getDesc(config)
    }, () => {});
};
</script>

<template>
    <div class="flex flex-col gap-4 min-w-0">
        <div v-for="(config, index) in filteredConfigs"
             :key="getKey(config) || index"
             class="bg-white/60 p-5 rounded-xl border border-slate-200 shadow-sm transition-all hover:shadow-md flex flex-col lg:flex-row gap-6">

            <div class="lg:w-1/3 flex flex-col gap-2 min-w-0">
                <h4 class="text-base font-bold text-slate-800 break-words">{{ getDesc(config) }}</h4>
                <div>
                    <span class="inline-flex items-center px-2 py-1 rounded-md text-xs font-mono font-bold bg-purple-50 text-purple-700 border border-purple-200/60 shadow-sm">
                        {{ getKey(config) }}
                    </span>
                </div>

                <div class="text-sm text-slate-500 mt-2 bg-white/50 p-2.5 rounded-lg border border-slate-100" v-if="getKey(config) === 'Transfer_BlockPeriod'">
                    請輸入「起始日-結束日」。<br>
                    例如：<code class="bg-slate-200 text-slate-800 px-1.5 py-0.5 rounded font-mono font-bold text-xs mx-0.5">25-1</code>
                </div>

                <div class="text-sm text-slate-500 mt-2 bg-white/50 p-2.5 rounded-lg border border-slate-100" v-if="getKey(config) === 'AssetNo_Prefixes'">
                    在此輸入多個 Prefix 加入清單。
                </div>
            </div>

            <div class="lg:w-2/3 flex flex-col justify-between min-w-0">
                <div v-if="getKey(config) !== 'AssetNo_Prefixes'" class="w-full">
                    <input type="text" v-model="config.configValue"
                           class="w-full lg:max-w-md px-4 py-2.5 bg-white border border-slate-300 rounded-lg font-mono text-sm focus:ring-2 focus:ring-purple-500/20 focus:border-purple-500 outline-none transition-all shadow-sm">
                </div>

                <div v-else class="flex flex-col gap-3 w-full">
                    <div class="flex items-center gap-2">
                        <input type="text" v-model="prefixInput" @keyup.enter="addPrefixToConfig(config)"
                               placeholder="輸入資產字首，例如 1656"
                               class="flex-1 lg:max-w-[240px] px-4 py-2.5 bg-white border border-slate-300 rounded-lg font-mono text-sm focus:ring-2 focus:ring-purple-500/20 focus:border-purple-500 outline-none transition-all shadow-sm">
                        <button @click="addPrefixToConfig(config)"
                                class="px-4 py-2.5 bg-white border border-slate-200 rounded-lg hover:bg-slate-50 text-slate-700 font-medium shadow-sm transition-all active:scale-95 text-sm whitespace-nowrap">
                            新增
                        </button>
                    </div>

                    <div class="bg-white/50 p-3 rounded-lg border border-slate-200 min-h-[3.25rem] flex flex-wrap gap-2 items-center">
                        <span v-for="p in parsePrefixes(getValue(config))" :key="p"
                              class="inline-flex items-center gap-1.5 px-3 py-1.5 rounded-md bg-white border border-slate-300 shadow-sm text-sm text-slate-700 font-mono font-bold group">
                            {{ p }}
                            <button class="text-slate-400 hover:text-red-600 hover:bg-red-50 rounded p-0.5 transition-colors flex items-center justify-center"
                                    title="移除" @click="removePrefixFromConfig(config, p)">
                                <svg xmlns="http://www.w3.org/2000/svg" class="w-3.5 h-3.5" fill="none" viewBox="0 0 24 24" stroke="currentColor" stroke-width="2">
                                    <path stroke-linecap="round" stroke-linejoin="round" d="M6 18L18 6M6 6l12 12" />
                                </svg>
                            </button>
                        </span>
                        <span v-if="parsePrefixes(getValue(config)).length === 0" class="text-sm text-slate-400 font-medium px-1">
                            目前尚未設定
                        </span>
                    </div>
                </div>

                <div class="mt-4 flex justify-end lg:justify-start">
                    <button @click="save(config)"
                            class="px-6 py-2.5 bg-purple-600 hover:bg-purple-700 text-white shadow-md shadow-purple-300/50 hover:shadow-lg rounded-lg transition-all transform active:scale-95 flex items-center gap-2 font-medium text-sm whitespace-nowrap">
                        儲存設定
                    </button>
                </div>
            </div>
        </div>
    </div>
</template>
