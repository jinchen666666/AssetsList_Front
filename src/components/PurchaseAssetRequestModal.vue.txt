<script setup>
    import { ref, computed, watch } from 'vue'
    import axios from 'axios'
    import UserPicker from './common/UserPicker.vue'

    const props = defineProps({
        visible: Boolean,
        currentUser: Object,
        selectedItems: {
            type: Array,
            default: () => []
        },
        systemConfigs: {
            type: Array,
            default: () => []
        },
        draftId: {
            type: [Number, String],
            default: null
        }
    })

    const emit = defineEmits(['close', 'saved'])

    const isSavingDraft = ref(false)
    const currentDraftId = ref(null)
    const isLoadingDraft = ref(false)
    const draftSourceItems = ref([])

    const form = ref({
        assetCategoryCode: '',
        itemCategoryCode: '',
        assetName: '',
        assetSpec: '',
        qty: 1,
        keeperId: '',
        keeperName: '',
        keeperDept: '',
        location: '',
        remark: ''
    })

    const effectiveSelectedItems = computed(() => {
        if (Array.isArray(props.selectedItems) && props.selectedItems.length > 0) {
            return props.selectedItems
        }
        return draftSourceItems.value
    })

    const getDisplayLocation = (item) => {
        return (
            item?.displayLocation ||
            item?.storageLocation ||
            item?.locationNo ||
            item?.warehouseName ||
            item?.warehouseNo ||
            item?.location ||
            ''
        )
    }

    const getSourceSummary = (item) => {
        const docTypeNo = item?.docTypeNo || ''
        const docNo = item?.docNo || ''
        const seq = item?.seq || ''
        const itemNo = item?.itemNo || ''
        const qty = item?.qty ?? ''

        return `${docTypeNo}-${docNo}-${seq}　品號:${itemNo}　數量:${qty}`
    }

    const getFirstSourceItem = () => {
        return effectiveSelectedItems.value?.[0] || {}
    }

    const publicSystemConfigs = ref([])

    const normalizePublicConfigs = (data) => {
        const d = data || {}

        const get = (...names) => {
            for (const name of names) {
                if (d[name] !== undefined && d[name] !== null) return d[name]
            }
            return ''
        }

        return [
            {
                configKey: 'AssetCode.Default.GeneralAffairs',
                configValue: get('defaultGeneralAffairs', 'DefaultGeneralAffairs')
            },
            {
                configKey: 'AssetCode.Default.AssetCategory',
                configValue: get('defaultAssetCategory', 'DefaultAssetCategory')
            },
            {
                configKey: 'AssetCode.Default.ItemCategory',
                configValue: get('defaultItemCategory', 'DefaultItemCategory')
            },
            {
                configKey: 'AssetCode.GeneralAffairsOptions',
                configValue: get('generalAffairsOptions', 'GeneralAffairsOptions')
            },
            {
                configKey: 'AssetCode.AssetCategoryOptions',
                configValue: get('assetCategoryOptions', 'AssetCategoryOptions')
            },
            {
                configKey: 'AssetCode.ItemCategoryOptions',
                configValue: get('itemCategoryOptions', 'ItemCategoryOptions')
            }
        ]
    }

    const loadPublicSystemConfigs = async () => {
        try {
            const res = await axios.get('/api/admin/config/public')
            publicSystemConfigs.value = normalizePublicConfigs(res.data || {})
        } catch (e) {
            console.error('讀取公開系統參數失敗:', e)
            publicSystemConfigs.value = []
        }
    }

    const effectiveSystemConfigs = computed(() => {
        if (Array.isArray(props.systemConfigs) && props.systemConfigs.length > 0) {
            return props.systemConfigs
        }

        return publicSystemConfigs.value
    })

    const getConfig = (key) =>
        (effectiveSystemConfigs.value || []).find(x => (x.configKey || x.ConfigKey) === key)

    const parseCodeOptions = (raw) => {
        const text = String(raw || '').trim()
        if (!text) return []

        // 支援格式：
        // 1. 04|家具
        // 2. 04家具
        // 3. 04家具 01工具 02庶電
        // 4. 04|家具,01|工具
        const tokens = text
            .split(/\r?\n|,|\s+/g)
            .map(x => x.trim())
            .filter(Boolean)

        return tokens
            .map(token => {
                if (token.includes('|')) {
                    const [codePart, namePart] = token.split('|', 2)
                    const code = String(codePart || '').trim().toUpperCase()
                    const name = String(namePart || '').trim()

                    return {
                        code,
                        name,
                        value: code,
                        text: name ? `${code} ${name}` : code,
                        label: name ? `${code} ${name}` : code
                    }
                }

                // 沒有 | 時，預設前兩碼是代碼，後面是名稱
                const match = token.match(/^([A-Za-z0-9]{2})(.*)$/)

                if (!match) {
                    const code = token.toUpperCase()

                    return {
                        code,
                        name: '',
                        value: code,
                        text: code,
                        label: code
                    }
                }

                const code = String(match[1] || '').trim().toUpperCase()
                const name = String(match[2] || '').trim()

                return {
                    code,
                    name,
                    value: code,
                    text: name ? `${code} ${name}` : code,
                    label: name ? `${code} ${name}` : code
                }
            })
            .filter(x => x.code)
    }

    const generalAffairsCode = computed(() =>
        String(
            getConfig('AssetCode.Default.GeneralAffairs')?.configValue ||
            getConfig('AssetCode.Default.GeneralAffairs')?.ConfigValue ||
            '99'
        ).trim().toUpperCase()
    )

    const assetCategoryOptions = computed(() =>
        parseCodeOptions(
            getConfig('AssetCode.AssetCategoryOptions')?.configValue ||
            getConfig('AssetCode.AssetCategoryOptions')?.ConfigValue ||
            ''
        )
    )

    const itemCategoryOptions = computed(() =>
        parseCodeOptions(
            getConfig('AssetCode.ItemCategoryOptions')?.configValue ||
            getConfig('AssetCode.ItemCategoryOptions')?.ConfigValue ||
            ''
        )
    )

    const incomingYearMonth = computed(() => {
        const first = effectiveSelectedItems.value.find(x => x?.incomingDate && String(x.incomingDate).length >= 8)
        if (!first) {
            const now = new Date()
            return now.toISOString().slice(2, 4) + String(now.getMonth() + 1).padStart(2, '0')
        }
        return String(first.incomingDate).substring(2, 6)
    })

    const assetCodePreview = computed(() => {
        const ga = (generalAffairsCode.value || '99').padStart(2, '0').substring(0, 2)
        const ac = String(form.value.assetCategoryCode || '').padStart(2, '0').substring(0, 2)
        const ic = String(form.value.itemCategoryCode || '').toUpperCase().padStart(2, '0').substring(0, 2)
        return `${ga}${ac}${ic}${incomingYearMonth.value}****`
    })

    const totalSourceQty = computed(() =>
        effectiveSelectedItems.value.reduce((sum, x) => sum + Number(x.qty || 0), 0)
    )

    const handleKeeperSelected = (user) => {
        form.value.keeperId = user?.id || ''
        form.value.keeperName = user?.name || user?.id || ''
        form.value.keeperDept = user?.deptName || user?.rawDeptName || user?.deptId || ''
    }

    const resetFormFromSelection = () => {
        const first = getFirstSourceItem()

        form.value = {
            assetCategoryCode: '',
            itemCategoryCode: '',
            assetName: first?.itemName || '',
            assetSpec: first?.itemSpec || '',
            qty: totalSourceQty.value || Number(first?.qty || 1),
            keeperId: first?.keeperId || first?.keeperNo || '',
            keeperName: first?.keeperName || '',
            keeperDept: first?.keeperDept || '',
            location: getDisplayLocation(first),
            remark: ''
        }

        currentDraftId.value = null
    }

    const loadDraft = async (draftId) => {
        if (!draftId) return

        isLoadingDraft.value = true
        try {
            const res = await axios.get(`/api/purchase-request-draft/${draftId}`)
            const draft = res.data
            if (!draft) return

            currentDraftId.value = draft.id || draftId

            const items = Array.isArray(draft.items) ? draft.items : []
            draftSourceItems.value = items.map(x => ({
                docTypeNo: x.docTypeNo || '',
                docNo: x.docNo || '',
                seq: x.seq || '',
                itemNo: x.itemNo || '',
                itemName: x.itemName || '',
                itemSpec: x.assetSpec || x.itemSpec || '',
                vendorName: x.vendorName || '',
                qty: x.qty || 1,
                incomingDate: x.acquireDate || '',
                location: x.location || '',
                selected: true
            }))

            const first = items.length > 0 ? items[0] : null
            if (!first) return

            form.value = {
                ...form.value,
                assetName: first.itemName || '',
                assetSpec: first.assetSpec || first.itemSpec || '',
                qty: first.qty || 1,
                keeperId: first.keeperId || '',
                keeperName: first.keeperName || '',
                keeperDept: first.keeperDept || '',
                location: first.location || '',
                remark: first.remarks || ''
            }
        } catch (e) {
            console.error(e)
            alert('載入待處理失敗')
        } finally {
            isLoadingDraft.value = false
        }
    }

    const saveDraft = async () => {
        if (!effectiveSelectedItems.value || effectiveSelectedItems.value.length === 0) {
            alert('沒有可儲存的來源資料')
            return
        }

        if (!form.value.assetName?.trim()) {
            alert('資產名稱不可空白')
            return
        }

        if (!form.value.keeperId?.trim()) {
            alert('請選擇保管人')
            return
        }

        if (!form.value.keeperDept?.trim()) {
            alert('請輸入保管部門')
            return
        }

        if (!form.value.location?.trim()) {
            alert('請輸入放置地點')
            return
        }

        isSavingDraft.value = true
        try {
            const first = effectiveSelectedItems.value[0] || {}

            const payload = {
                id: currentDraftId.value,
                requestType: 'SingleAsset',
                status: 'Draft',
                sourceDocTypeNo: first.docTypeNo || '',
                sourceDocNo: first.docNo || '',
                sourceDocSummary: `建立單一資產 (${effectiveSelectedItems.value.length} 筆來源資料)`,
                remarks: form.value.remark || '',
                items: [
                    {
                        docTypeNo: first.docTypeNo || '',
                        docNo: first.docNo || '',
                        seq: effectiveSelectedItems.value.map(x => x.seq).filter(Boolean).join(','),
                        itemNo: first.itemNo || '',
                        itemName: form.value.assetName || '',
                        assetSpec: form.value.assetSpec || '',
                        vendorId: first.vendorId || '',
                        vendorName: first.vendorName || '',
                        qty: Number(form.value.qty || 1),
                        previewAssetNo: assetCodePreview.value,
                        keeperId: form.value.keeperId || '',
                        keeperName: form.value.keeperName || '',
                        keeperDept: form.value.keeperDept || '',
                        location: form.value.location || '',
                        acquireDate: first.incomingDate || '',
                        unit: first.unit || '',
                        remarks: form.value.remark || '',
                        sortOrder: 1
                    }
                ]
            }

            const res = await axios.post('/api/purchase-request-draft/save', payload, {
                headers: {
                    'X-User-Id': props.currentUser?.userId || props.currentUser?.id || '',
                    'X-User-Name': encodeURIComponent(props.currentUser?.userName || props.currentUser?.name || '')
                }
            })

            if (res?.data?.success) {
                currentDraftId.value = res.data.id
                alert('待處理已儲存')
                emit('saved')
            } else {
                alert('待處理儲存失敗')
            }
        } catch (e) {
            console.error(e)
            alert('待處理儲存失敗')
        } finally {
            isSavingDraft.value = false
        }
    }

    watch(
        () => [props.visible, props.draftId],
        async ([visible, draftId]) => {
            if (!visible) return

            await loadPublicSystemConfigs()

            draftSourceItems.value = []

            if (draftId) {
                await loadDraft(draftId)
            } else {
                resetFormFromSelection()
            }
        },
        { immediate: true }
    )
</script>

<template>
    <div v-if="visible" class="fixed inset-0 z-50 bg-slate-900/40 backdrop-blur-sm flex items-center justify-center p-4">
        <div class="w-full max-w-4xl bg-white/95 backdrop-blur-xl rounded-2xl border border-white/60 shadow-2xl overflow-hidden">
            <div class="px-6 py-4 border-b border-slate-200/50 bg-slate-50/50 flex items-center justify-between">
                <h3 class="text-lg font-bold text-slate-800">建立單一資產</h3>
                <button @click="$emit('close')" class="text-slate-400 hover:text-slate-700 transition-colors">✕</button>
            </div>

            <div class="p-6 space-y-5 max-h-[80vh] overflow-auto">
                <div v-if="isLoadingDraft" class="p-6 text-center text-slate-500 font-medium">
                    載入待處理中...
                </div>

                <template v-else>
                    <div class="bg-indigo-50/60 border border-indigo-100 rounded-xl p-4 shadow-sm">
                        <div class="text-sm text-indigo-700 mb-3 font-bold flex items-center gap-2">
                            來源進貨單資訊
                        </div>
                        <div class="space-y-2">
                            <div v-for="(item, i) in effectiveSelectedItems"
                                 :key="`${item.docNo}-${item.seq}-${i}`"
                                 class="text-sm text-slate-700 font-medium">
                                {{ getSourceSummary(item) }}
                            </div>
                        </div>
                    </div>

                    <div class="grid grid-cols-1 md:grid-cols-2 gap-4">
                        <div>
                            <label class="block text-sm font-bold text-slate-700 mb-2">資產類別</label>
                            <select v-model="form.assetCategoryCode" class="w-full px-4 py-2.5 border border-slate-300 rounded-lg bg-white/80 text-sm focus:ring-2 focus:ring-indigo-500/20 focus:border-indigo-500 transition-all shadow-sm">
                                <option value="">請選擇</option>
                                <option v-for="opt in assetCategoryOptions" :key="opt.code" :value="opt.code">
                                    {{ opt.code }} - {{ opt.name }}
                                </option>
                            </select>
                        </div>

                        <div>
                            <label class="block text-sm font-bold text-slate-700 mb-2">品項類別</label>
                            <select v-model="form.itemCategoryCode" class="w-full px-4 py-2.5 border border-slate-300 rounded-lg bg-white/80 text-sm focus:ring-2 focus:ring-indigo-500/20 focus:border-indigo-500 transition-all shadow-sm">
                                <option value="">請選擇</option>
                                <option v-for="opt in itemCategoryOptions" :key="opt.code" :value="opt.code">
                                    {{ opt.code }} - {{ opt.name }}
                                </option>
                            </select>
                        </div>

                        <div>
                            <label class="block text-sm font-bold text-slate-700 mb-2">YYMM</label>
                            <input :value="incomingYearMonth" disabled class="w-full px-4 py-2.5 border border-slate-200 rounded-lg bg-slate-50/80 text-slate-500 text-sm" />
                        </div>

                        <div>
                            <label class="block text-sm font-bold text-slate-700 mb-2">編碼預覽</label>
                            <input :value="assetCodePreview" disabled class="w-full px-4 py-2.5 border border-slate-200 rounded-lg bg-slate-50/80 text-slate-500 text-sm font-mono font-medium" />
                        </div>

                        <div class="md:col-span-2">
                            <label class="block text-sm font-bold text-slate-700 mb-2">資產名稱</label>
                            <input v-model="form.assetName"
                                   readonly
                                   class="w-full px-4 py-2.5 border border-slate-300 rounded-lg bg-slate-50/80 text-sm text-slate-600 cursor-not-allowed shadow-sm" />
                        </div>

                        <div class="md:col-span-2">
                            <label class="block text-sm font-bold text-slate-700 mb-2">資產規格</label>
                            <input v-model="form.assetSpec"
                                   readonly
                                   class="w-full px-4 py-2.5 border border-slate-300 rounded-lg bg-slate-50/80 text-sm text-slate-600 cursor-not-allowed shadow-sm" />
                        </div>

                        <div>
                            <label class="block text-sm font-bold text-slate-700 mb-2">數量</label>
                            <input v-model.number="form.qty" type="number" min="1" class="w-full px-4 py-2.5 border border-slate-300 rounded-lg bg-white/80 text-sm focus:ring-2 focus:ring-indigo-500/20 focus:border-indigo-500 transition-all shadow-sm" />
                            <div class="text-xs text-indigo-500 mt-1.5 font-medium">來源總數量：{{ totalSourceQty }}</div>
                        </div>

                        <UserPicker v-model:userId="form.keeperId"
                                    v-model:userName="form.keeperName"
                                    v-model:deptName="form.keeperDept"
                                    label="保管人"
                                    placeholder="搜尋保管人工號或姓名"
                                    required
                                    @selected="handleKeeperSelected" />

                        <div>
                            <label class="block text-sm font-bold text-slate-700 mb-2">保管部門</label>
                            <input v-model="form.keeperDept" readonly class="w-full px-4 py-2.5 border border-slate-300 rounded-lg bg-slate-50/80 text-sm text-slate-600 cursor-not-allowed shadow-sm" />
                        </div>

                        <div>
                            <label class="block text-sm font-bold text-slate-700 mb-2">放置地點</label>
                            <input v-model="form.location" class="w-full px-4 py-2.5 border border-slate-300 rounded-lg bg-white/80 text-sm focus:ring-2 focus:ring-indigo-500/20 focus:border-indigo-500 transition-all shadow-sm" />
                        </div>

                        <div class="md:col-span-2">
                            <label class="block text-sm font-bold text-slate-700 mb-2">備註</label>
                            <textarea v-model="form.remark" rows="3" class="w-full px-4 py-2.5 border border-slate-300 rounded-lg bg-white/80 text-sm focus:ring-2 focus:ring-indigo-500/20 focus:border-indigo-500 transition-all shadow-sm"></textarea>
                        </div>
                    </div>
                </template>
            </div>

            <div class="px-6 py-4 border-t border-slate-200/50 bg-slate-50/50 flex justify-end gap-3">
                <button @click="$emit('close')" class="px-4 py-2 rounded-lg border border-slate-300 bg-white hover:bg-slate-50 text-slate-700 transition-all transform active:scale-95 shadow-sm font-medium">
                    取消
                </button>

                <button @click="saveDraft"
                        :disabled="isSavingDraft || isLoadingDraft"
                        class="px-4 py-2 bg-white/80 hover:bg-white text-slate-700 rounded-lg shadow-sm border border-slate-200 transition-all flex items-center gap-2 hover:shadow-md transform active:scale-95 font-medium disabled:opacity-60">
                    <span v-if="isSavingDraft" class="animate-spin border-2 border-slate-400 border-t-transparent rounded-full w-4 h-4"></span>
                    儲存待處理
                </button>
            </div>
        </div>
    </div>
</template>
