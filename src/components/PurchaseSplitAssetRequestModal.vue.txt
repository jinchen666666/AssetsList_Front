<script setup>
    import { ref, computed, watch } from 'vue'
    import axios from 'axios'
    import UserPicker from './common/UserPicker.vue'

    const props = defineProps({
        visible: Boolean,
        currentUser: Object,
        selectedItems: {
            type: Array,
            default: () => []
        },
        systemConfigs: {
            type: Array,
            default: () => []
        },
        draftId: {
            type: [Number, String],
            default: null
        }
    })

    const emit = defineEmits(['close', 'saved'])

    const isSavingDraft = ref(false)
    const currentDraftId = ref(null)
    const isLoadingDraft = ref(false)
    const draftSourceItems = ref([])

    const splitRows = ref([])

    const publicSystemConfigs = ref([])

    const normalizePublicConfigs = (data) => {
        const d = data || {}

        const get = (...names) => {
            for (const name of names) {
                if (d[name] !== undefined && d[name] !== null) return d[name]
            }
            return ''
        }

        return [
            {
                configKey: 'AssetCode.Default.GeneralAffairs',
                configValue: get('defaultGeneralAffairs', 'DefaultGeneralAffairs')
            },
            {
                configKey: 'AssetCode.Default.AssetCategory',
                configValue: get('defaultAssetCategory', 'DefaultAssetCategory')
            },
            {
                configKey: 'AssetCode.Default.ItemCategory',
                configValue: get('defaultItemCategory', 'DefaultItemCategory')
            },
            {
                configKey: 'AssetCode.GeneralAffairsOptions',
                configValue: get('generalAffairsOptions', 'GeneralAffairsOptions')
            },
            {
                configKey: 'AssetCode.AssetCategoryOptions',
                configValue: get('assetCategoryOptions', 'AssetCategoryOptions')
            },
            {
                configKey: 'AssetCode.ItemCategoryOptions',
                configValue: get('itemCategoryOptions', 'ItemCategoryOptions')
            }
        ]
    }

    const loadPublicSystemConfigs = async () => {
        try {
            const res = await axios.get('/api/admin/config/public')
            publicSystemConfigs.value = normalizePublicConfigs(res.data || {})
        } catch (e) {
            console.error('讀取公開系統參數失敗:', e)
            publicSystemConfigs.value = []
        }
    }

    const effectiveSystemConfigs = computed(() => {
        if (Array.isArray(props.systemConfigs) && props.systemConfigs.length > 0) {
            return props.systemConfigs
        }

        return publicSystemConfigs.value
    })

    const getConfig = (key) =>
        (effectiveSystemConfigs.value || []).find(x => (x.configKey || x.ConfigKey) === key)

    const parseCodeOptions = (raw) => {
        const text = String(raw || '').trim()
        if (!text) return []

        // 支援格式：
        // 1. 04|家具
        // 2. 04家具
        // 3. 04家具 01工具 02庶電
        // 4. 04|家具,01|工具
        const tokens = text
            .split(/\r?\n|,|\s+/g)
            .map(x => x.trim())
            .filter(Boolean)

        return tokens
            .map(token => {
                if (token.includes('|')) {
                    const [codePart, namePart] = token.split('|', 2)
                    const code = String(codePart || '').trim().toUpperCase()
                    const name = String(namePart || '').trim()

                    return {
                        code,
                        name,
                        value: code,
                        text: name ? `${code} ${name}` : code,
                        label: name ? `${code} ${name}` : code
                    }
                }

                // 沒有 | 時，預設前兩碼是代碼，後面是名稱
                const match = token.match(/^([A-Za-z0-9]{2})(.*)$/)

                if (!match) {
                    const code = token.toUpperCase()

                    return {
                        code,
                        name: '',
                        value: code,
                        text: code,
                        label: code
                    }
                }

                const code = String(match[1] || '').trim().toUpperCase()
                const name = String(match[2] || '').trim()

                return {
                    code,
                    name,
                    value: code,
                    text: name ? `${code} ${name}` : code,
                    label: name ? `${code} ${name}` : code
                }
            })
            .filter(x => x.code)
    }

    const generalAffairsCode = computed(() =>
        String(
            getConfig('AssetCode.Default.GeneralAffairs')?.configValue ||
            getConfig('AssetCode.Default.GeneralAffairs')?.ConfigValue ||
            '99'
        ).trim().toUpperCase()
    )

    const assetCategoryOptions = computed(() =>
        parseCodeOptions(
            getConfig('AssetCode.AssetCategoryOptions')?.configValue ||
            getConfig('AssetCode.AssetCategoryOptions')?.ConfigValue ||
            ''
        )
    )

    const itemCategoryOptions = computed(() =>
        parseCodeOptions(
            getConfig('AssetCode.ItemCategoryOptions')?.configValue ||
            getConfig('AssetCode.ItemCategoryOptions')?.ConfigValue ||
            ''
        )
    )

    const effectiveSelectedItems = computed(() => {
        if (Array.isArray(props.selectedItems) && props.selectedItems.length > 0) {
            return props.selectedItems
        }
        return draftSourceItems.value
    })

    const getDisplayLocation = (item) => {
        return (
            item?.displayLocation ||
            item?.storageLocation ||
            item?.locationNo ||
            item?.warehouseName ||
            item?.warehouseNo ||
            item?.location ||
            ''
        )
    }

    const getSourceSummary = (item) => {
        const docTypeNo = item?.docTypeNo || ''
        const docNo = item?.docNo || ''
        const seq = item?.seq || ''
        const itemNo = item?.itemNo || ''
        const qty = item?.qty ?? ''

        return `${docTypeNo}-${docNo}-${seq}　品號:${itemNo}　數量:${qty}`
    }

    const getFirstSourceItem = () => {
        return effectiveSelectedItems.value?.[0] || {}
    }

    const totalSourceQty = computed(() =>
        effectiveSelectedItems.value.reduce((sum, x) => sum + Number(x.qty || 0), 0)
    )

    const splitTotalQty = computed(() =>
        splitRows.value.reduce((sum, row) => sum + Number(row.qty || 0), 0)
    )

    const qtyExceeded = computed(() => splitTotalQty.value > totalSourceQty.value)

    const qtyMatched = computed(() => splitTotalQty.value === totalSourceQty.value)

    const qtyHintText = computed(() => {
        if (qtyMatched.value) return '拆分總數量與來源總數量一致'
        if (qtyExceeded.value) return '拆分總數量大於來源總數量，請確認是否仍要儲存'
        return '拆分總數量小於來源總數量，請確認是否仍要儲存'
    })

    const qtyHintClass = computed(() => {
        if (qtyMatched.value) return 'text-emerald-600'
        return 'text-amber-600'
    })

    const qtyTotalClass = computed(() => {
        if (qtyMatched.value) return 'text-emerald-700'
        return 'text-amber-700'
    })

    const getIncomingYearMonth = (incomingDate) => {
        if (incomingDate && String(incomingDate).length >= 8) {
            return String(incomingDate).substring(2, 6)
        }
        const now = new Date()
        return now.toISOString().slice(2, 4) + String(now.getMonth() + 1).padStart(2, '0')
    }

    const getRowPreviewCode = (row) => {
        const ga = (generalAffairsCode.value || '99').padStart(2, '0').substring(0, 2)
        const ac = String(row.assetCategoryCode || '').padStart(2, '0').substring(0, 2)
        const ic = String(row.itemCategoryCode || '').toUpperCase().padStart(2, '0').substring(0, 2)
        const yymm = getIncomingYearMonth(row.incomingDate)
        return `${ga}${ac}${ic}${yymm}****`
    }

    const handleSplitKeeperSelected = (row, user) => {
        row.keeperId = user?.id || ''
        row.keeperName = user?.name || user?.id || ''
        row.keeperDept = user?.deptName || user?.rawDeptName || user?.deptId || ''
    }

    const createDefaultSplitRow = (sourceItem, index = 0) => {
        const first = sourceItem && Object.keys(sourceItem).length > 0
            ? sourceItem
            : getFirstSourceItem()

        return {
            id: `${Date.now()}_${index}_${Math.random().toString(36).slice(2, 8)}`,
            sourceSeq: first?.seq || '',
            assetCategoryCode: '',
            itemCategoryCode: '',
            assetName: first?.itemName || '',
            assetSpec: first?.itemSpec || '',
            vendorId: first?.vendorId || '',
            vendorName: first?.vendorName || '',
            qty: Number(first?.qty || 1),
            keeperId: first?.keeperId || first?.keeperNo || '',
            keeperName: first?.keeperName || '',
            keeperDept: first?.keeperDept || '',
            location: getDisplayLocation(first),
            remark: '',
            incomingDate: first?.incomingDate || ''
        }
    }

    const resetRowsFromSelection = () => {
        splitRows.value = effectiveSelectedItems.value.map((item, index) =>
            createDefaultSplitRow(item, index)
        )

        if (splitRows.value.length === 0) {
            splitRows.value = [createDefaultSplitRow({}, 0)]
        }

        currentDraftId.value = null
    }

    const addSplitRow = () => {
        const sourceBase = effectiveSelectedItems.value[0] || {}
        splitRows.value.push(createDefaultSplitRow(sourceBase, splitRows.value.length))
    }

    const removeSplitRow = (rowIndex) => {
        if (splitRows.value.length <= 1) {
            alert('至少需保留一筆拆分資料')
            return
        }
        splitRows.value.splice(rowIndex, 1)
    }

    const loadDraft = async (draftId) => {
        if (!draftId) return

        isLoadingDraft.value = true
        try {
            const res = await axios.get(`/api/purchase-request-draft/${draftId}`)
            const draft = res.data
            if (!draft) return

            currentDraftId.value = draft.id || draftId

            const items = Array.isArray(draft.items) ? draft.items : []

            draftSourceItems.value = items.map(x => ({
                docTypeNo: x.docTypeNo || '',
                docNo: x.docNo || '',
                seq: x.seq || '',
                itemNo: x.itemNo || '',
                itemName: x.itemName || '',
                itemSpec: x.assetSpec || x.itemSpec || '',
                vendorName: x.vendorName || '',
                qty: x.qty || 1,
                incomingDate: x.acquireDate || '',
                location: x.location || '',
                selected: true
            }))

            splitRows.value = items.map((x, index) => ({
                id: `${Date.now()}_${index}_${Math.random().toString(36).slice(2, 8)}`,
                sourceSeq: x.seq || '',
                assetCategoryCode: '',
                itemCategoryCode: '',
                assetName: x.itemName || '',
                assetSpec: x.assetSpec || x.itemSpec || '',
                vendorName: x.vendorName || '',
                qty: Number(x.qty || 1),
                keeperId: x.keeperId || '',
                keeperName: x.keeperName || '',
                keeperDept: x.keeperDept || '',
                location: x.location || '',
                remark: x.remarks || '',
                incomingDate: x.acquireDate || ''
            }))

            if (splitRows.value.length === 0) {
                splitRows.value = [createDefaultSplitRow({}, 0)]
            }
        } catch (e) {
            console.error(e)
            alert('載入待處理失敗')
        } finally {
            isLoadingDraft.value = false
        }
    }

    const validateBeforeSave = () => {
        if (splitRows.value.length === 0) {
            alert('沒有可儲存的拆分資料')
            return false
        }

        if (splitTotalQty.value <= 0) {
            alert('拆分數量需大於 0')
            return false
        }

        for (const row of splitRows.value) {
            if (!row.assetCategoryCode) {
                alert('請選擇資產類別')
                return false
            }
            if (!row.itemCategoryCode) {
                alert('請選擇品項類別')
                return false
            }
            if (!row.assetName?.trim()) {
                alert('資產名稱不可空白')
                return false
            }
            if (!row.qty || Number(row.qty) <= 0) {
                alert('數量需大於 0')
                return false
            }
            if (!row.keeperId?.trim()) {
                alert('請選擇保管人')
                return false
            }
            if (!row.keeperDept?.trim()) {
                alert('請輸入保管部門')
                return false
            }
            if (!row.location?.trim()) {
                alert('請輸入放置地點')
                return false
            }
        }

        return true
    }

    const saveDraft = async () => {
        if (!effectiveSelectedItems.value || effectiveSelectedItems.value.length === 0) {
            alert('沒有可儲存的來源資料')
            return
        }

        if (!validateBeforeSave()) return

        if (!qtyMatched.value) {
            const confirmed = confirm('拆分總數量與來源總數量不一致，是否仍要儲存？')
            if (!confirmed) return
        }

        isSavingDraft.value = true
        try {
            const first = effectiveSelectedItems.value[0] || {}

            const payload = {
                id: currentDraftId.value,
                requestType: 'SplitAsset',
                status: 'Draft',
                sourceDocTypeNo: first.docTypeNo || '',
                sourceDocNo: first.docNo || '',
                sourceDocSummary: `拆為多筆資產 (${splitRows.value.length} 筆拆分資料)`,
                remarks: '',
                items: splitRows.value.map((row, index) => ({
                    docTypeNo: first.docTypeNo || '',
                    docNo: first.docNo || '',
                    seq: row.sourceSeq || first.seq || '',
                    itemNo: first.itemNo || '',
                    itemName: row.assetName || '',
                    assetSpec: row.assetSpec || '',
                    vendorId: row.vendorId || first.vendorId || '',
                    vendorName: row.vendorName || first.vendorName || '',
                    qty: Number(row.qty || 1),
                    previewAssetNo: getRowPreviewCode(row),
                    keeperId: row.keeperId || '',
                    keeperName: row.keeperName || '',
                    keeperDept: row.keeperDept || '',
                    location: row.location || '',
                    acquireDate: row.incomingDate || first.incomingDate || '',
                    unit: first.unit || '',
                    remarks: row.remark || '',
                    sortOrder: index + 1
                }))
            }

            const res = await axios.post('/api/purchase-request-draft/save', payload, {
                headers: {
                    'X-User-Id': props.currentUser?.userId || props.currentUser?.id || '',
                    'X-User-Name': encodeURIComponent(props.currentUser?.userName || props.currentUser?.name || '')
                }
            })

            if (res?.data?.success) {
                currentDraftId.value = res.data.id
                alert('待處理已儲存')
                emit('saved')
            } else {
                alert('待處理儲存失敗')
            }
        } catch (e) {
            console.error(e)
            alert('待處理儲存失敗')
        } finally {
            isSavingDraft.value = false
        }
    }

    watch(
        () => [props.visible, props.draftId],
        async ([visible, draftId]) => {
            if (!visible) return

            await loadPublicSystemConfigs()

            draftSourceItems.value = []

            if (draftId) {
                await loadDraft(draftId)
            } else {
                resetRowsFromSelection()
            }
        },
        { immediate: true }
    )
</script>

<template>
    <div v-if="visible" class="fixed inset-0 z-50 bg-slate-900/40 backdrop-blur-sm flex items-center justify-center p-4">
        <div class="w-full max-w-6xl bg-white/95 backdrop-blur-xl rounded-2xl border border-white/60 shadow-2xl overflow-hidden">
            <div class="px-6 py-4 border-b border-slate-200/50 bg-slate-50/50 flex items-center justify-between">
                <h3 class="text-lg font-bold text-slate-800">拆為多筆資產</h3>
                <button @click="$emit('close')" class="text-slate-400 hover:text-slate-700 transition-colors">✕</button>
            </div>

            <div class="p-6 space-y-5 max-h-[80vh] overflow-auto">
                <div v-if="isLoadingDraft" class="p-6 text-center text-slate-500 font-medium">
                    載入待處理中...
                </div>

                <template v-else>
                    <div class="bg-indigo-50/60 border border-indigo-100 rounded-xl p-4 shadow-sm">
                        <div class="text-sm text-indigo-700 mb-3 font-bold">來源進貨單資訊</div>
                        <div class="space-y-2">
                            <div v-for="(item, i) in effectiveSelectedItems"
                                 :key="`${item.docNo}-${item.seq}-${i}`"
                                 class="text-sm text-slate-700 font-medium">
                                {{ getSourceSummary(item) }}
                            </div>
                        </div>
                    </div>

                    <div class="flex justify-between items-center">
                        <div>
                            <div class="text-sm text-slate-500">
                                來源總數量：
                                <span class="font-bold text-slate-700">{{ totalSourceQty }}</span>
                                <span class="mx-2 text-slate-300">/</span>
                                拆分總數量：
                                <span class="font-bold" :class="qtyTotalClass">
                                    {{ splitTotalQty }}
                                </span>
                            </div>
                            <div class="mt-1 text-xs font-medium" :class="qtyHintClass">
                                {{ qtyHintText }}
                            </div>
                        </div>

                        <button @click="addSplitRow"
                                class="px-4 py-2 bg-white/80 hover:bg-white text-slate-700 rounded-lg shadow-sm border border-slate-200 transition-all font-medium">
                            新增拆分列
                        </button>
                    </div>

                    <div class="space-y-4">
                        <div v-for="(row, rowIndex) in splitRows" :key="row.id" class="border border-slate-200 rounded-xl bg-white/70 p-4">
                            <div class="flex justify-between items-center mb-4">
                                <div class="font-bold text-slate-700">拆分列 {{ rowIndex + 1 }}</div>
                                <button @click="removeSplitRow(rowIndex)"
                                        class="px-3 py-1.5 bg-rose-50 hover:bg-rose-100 text-rose-600 border border-rose-200 rounded-lg shadow-sm transition-all font-medium">
                                    刪除
                                </button>
                            </div>

                            <div class="grid grid-cols-1 md:grid-cols-2 gap-4">
                                <div>
                                    <label class="block text-sm font-bold text-slate-700 mb-2">資產類別</label>
                                    <select v-model="row.assetCategoryCode" class="w-full px-4 py-2.5 border border-slate-300 rounded-lg bg-white/80 text-sm">
                                        <option value="">請選擇</option>
                                        <option v-for="opt in assetCategoryOptions" :key="opt.code" :value="opt.code">
                                            {{ opt.code }} - {{ opt.name }}
                                        </option>
                                    </select>
                                </div>

                                <div>
                                    <label class="block text-sm font-bold text-slate-700 mb-2">品項類別</label>
                                    <select v-model="row.itemCategoryCode" class="w-full px-4 py-2.5 border border-slate-300 rounded-lg bg-white/80 text-sm">
                                        <option value="">請選擇</option>
                                        <option v-for="opt in itemCategoryOptions" :key="opt.code" :value="opt.code">
                                            {{ opt.code }} - {{ opt.name }}
                                        </option>
                                    </select>
                                </div>

                                <div>
                                    <label class="block text-sm font-bold text-slate-700 mb-2">編碼預覽</label>
                                    <input :value="getRowPreviewCode(row)" disabled class="w-full px-4 py-2.5 border border-slate-200 rounded-lg bg-slate-50/80 text-slate-500 text-sm font-mono font-medium" />
                                </div>

                                <div>
                                    <label class="block text-sm font-bold text-slate-700 mb-2">數量</label>
                                    <input v-model.number="row.qty" type="number" min="1" class="w-full px-4 py-2.5 border border-slate-300 rounded-lg bg-white/80 text-sm" />
                                </div>

                                <div class="md:col-span-2">
                                    <label class="block text-sm font-bold text-slate-700 mb-2">資產名稱</label>
                                    <input v-model="row.assetName"
                                           readonly
                                           class="w-full px-4 py-2.5 border border-slate-300 rounded-lg bg-slate-50/80 text-sm text-slate-600 cursor-not-allowed" />
                                </div>

                                <div class="md:col-span-2">
                                    <label class="block text-sm font-bold text-slate-700 mb-2">資產規格</label>
                                    <input v-model="row.assetSpec"
                                           readonly
                                           class="w-full px-4 py-2.5 border border-slate-300 rounded-lg bg-slate-50/80 text-sm text-slate-600 cursor-not-allowed" />
                                </div>

                                <UserPicker v-model:userId="row.keeperId"
                                            v-model:userName="row.keeperName"
                                            v-model:deptName="row.keeperDept"
                                            label="保管人"
                                            placeholder="搜尋保管人工號或姓名"
                                            required
                                            @selected="handleSplitKeeperSelected(row, $event)" />

                                <div>
                                    <label class="block text-sm font-bold text-slate-700 mb-2">保管部門</label>
                                    <input v-model="row.keeperDept" readonly class="w-full px-4 py-2.5 border border-slate-300 rounded-lg bg-slate-50/80 text-sm text-slate-600 cursor-not-allowed" />
                                </div>

                                <div>
                                    <label class="block text-sm font-bold text-slate-700 mb-2">放置地點</label>
                                    <input v-model="row.location" class="w-full px-4 py-2.5 border border-slate-300 rounded-lg bg-white/80 text-sm" />
                                </div>

                                <div class="md:col-span-2">
                                    <label class="block text-sm font-bold text-slate-700 mb-2">備註</label>
                                    <textarea v-model="row.remark" rows="2" class="w-full px-4 py-2.5 border border-slate-300 rounded-lg bg-white/80 text-sm"></textarea>
                                </div>
                            </div>
                        </div>
                    </div>
                </template>
            </div>

            <div class="px-6 py-4 border-t border-slate-200/50 bg-slate-50/50 flex justify-end gap-3">
                <button @click="$emit('close')"
                        class="px-4 py-2 rounded-lg border border-slate-300 bg-white hover:bg-slate-50 text-slate-700 transition-all transform active:scale-95 shadow-sm font-medium">
                    取消
                </button>

                <button @click="saveDraft"
                        :disabled="isSavingDraft || isLoadingDraft"
                        class="px-4 py-2 bg-white/80 hover:bg-white text-slate-700 rounded-lg shadow-sm border border-slate-200 transition-all flex items-center gap-2 hover:shadow-md transform active:scale-95 font-medium disabled:opacity-60">
                    <span v-if="isSavingDraft" class="animate-spin border-2 border-slate-400 border-t-transparent rounded-full w-4 h-4"></span>
                    儲存待處理
                </button>
            </div>
        </div>
    </div>
</template>
